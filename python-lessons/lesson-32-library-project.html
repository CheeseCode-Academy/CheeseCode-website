🎯 هدف این درس

توی این درس، یک پروژه‌ی کامل برای مدیریت کتابخانه می‌سازیم که همه‌چیزهایی که تا حالا یاد گرفتی رو به کار می‌گیره.

📖 پروژه چیست؟

این پروژه شامل:

  • مدیریت کتاب‌ها (اضافه، حذف، جستجو)
  • مدیریت اعضا (ثبت نام، امانت)
  • امانت و بازگشت کتاب
  • ذخیره اطلاعات در فایل

💻 کلاس کتاب

class کتاب:
    def __init__(self, عنوان, نویسنده, سال, شابک):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
        self.سال = سال
        self.شابک = شابک
        self.امانت_رفته = False
        self.امانت_گیرنده = None
    
    def امانت_بده(self, عضو):
        if self.امانت_رفته:
            return "کتاب قبلاً امانت رفته است!"
        self.امانت_رفته = True
        self.امانت_گیرنده = عضو
        return f"کتاب '{self.عنوان}' به {عضو.نام} امانت داده شد"
    
    def بازگردان(self):
        if not self.امانت_رفته:
            return "کتاب در کتابخانه است!"
        self.امانت_رفته = False
        self.امانت_گیرنده = None
        return f"کتاب '{self.عنوان}' بازگردانده شد"
    
    def __str__(self):
        وضعیت = "📕 موجود" if not self.امانت_رفته else f"📗 امانت: {self.امانت_گیرنده.نام if self.امانت_گیرنده else 'نامشخص'}"
        return f"{self.عنوان} - {self.نویسنده} ({self.سال}) [{وضعیت}]"

💻 کلاس عضو

class عضو:
    def __init__(self, نام, شماره_عضو):
        self.نام = نام
        self.شماره_عضو = شماره_عضو
        self.کتاب‌های_امانت = []
        self.تعداد_مجاز = 3
    
    def می_تواند_امانت_بگیرد(self):
        return len(self.کتاب‌های_امانت) < self.تعداد_مجاز
    
    def امانت_گرفتن(self, کتاب):
        if not self.می_تواند_امانت_بگیرد():
            return f"{self.نام} نمی‌تواند بیشتر از {self.تعداد_مجاز} کتاب امانت بگیرد!"
        if کتاب.امانت_رفته:
            return "کتاب قبلاً امانت رفته است!"
        کتاب.امانت_بده(self)
        self.کتاب‌های_امانت.append(کتاب)
        return f"{self.نام} کتاب '{کتاب.عنوان}' را امانت گرفت"
    
    def بازگرداندن(self, کتاب):
        if کتاب not in self.کتاب‌های_امانت:
            return f"{self.نام} این کتاب را امانت نگرفته است!"
        کتاب.بازگردان()
        self.کتاب‌های_امانت.remove(کتاب)
        return f"{self.نام} کتاب '{کتاب.عنوان}' را بازگرداند"
    
    def __str__(self):
        return f"{self.نام} (شماره: {self.شماره_عضو}) - {len(self.کتاب‌های_امانت)} کتاب امانت"

💻 کلاس کتابخانه

class کتابخانه:
    def __init__(self, نام):
        self.نام = نام
        self.کتاب‌ها = []
        self.اعضا = []
    
    def اضافه_کتاب(self, کتاب):
        self.کتاب‌ها.append(کتاب)
        return f"کتاب '{کتاب.عنوان}' به کتابخانه اضافه شد"
    
    def حذف_کتاب(self, شابک):
        for کتاب in self.کتاب‌ها:
            if کتاب.شابک == شابک:
                if کتاب.امانت_رفته:
                    return "کتاب امانت رفته است، نمی‌توان حذف کرد!"
                self.کتاب‌ها.remove(کتاب)
                return f"کتاب '{کتاب.عنوان}' حذف شد"
        return "کتاب پیدا نشد!"
    
    def ثبت_عضو(self, عضو):
        self.اعضا.append(عضو)
        return f"عضو '{عضو.نام}' ثبت شد"
    
    def جستجو_کتاب(self, عبارت):
        نتایج = []
        for کتاب in self.کتاب‌ها:
            if عبارت.lower() in کتاب.عنوان.lower() or عبارت.lower() in کتاب.نویسنده.lower():
                نتایج.append(کتاب)
        return نتایج
    
    def کتاب‌های_موجود(self):
        return [کتاب for کتاب in self.کتاب‌ها if not کتاب.امانت_رفته]
    
    def نمایش_کتاب‌ها(self):
        if not self.کتاب‌ها:
            return "هیچ کتابی در کتابخانه نیست!"
        result = "📚 لیست کتاب‌ها:\n"
        for کتاب in self.کتاب‌ها:
            result += f"  - {کتاب}\n"
        return result
    
    def امانت_کتاب(self, شماره_عضو, شابک):
        عضو = self._پیدا_کن_عضو(شماره_عضو)
        if not عضو:
            return "عضو پیدا نشد!"
        
        کتاب = self._پیدا_کن_کتاب(شابک)
        if not کتاب:
            return "کتاب پیدا نشد!"
        
        return عضو.امانت_گرفتن(کتاب)
    
    def بازگرداندن_کتاب(self, شماره_عضو, شابک):
        عضو = self._پیدا_کن_عضو(شماره_عضو)
        if not عضو:
            return "عضو پیدا نشد!"
        
        کتاب = self._پیدا_کن_کتاب(شابک)
        if not کتاب:
            return "کتاب پیدا نشد!"
        
        return عضو.بازگرداندن(کتاب)
    
    def _پیدا_کن_عضو(self, شماره):
        for عضو in self.اعضا:
            if عضو.شماره_عضو == شماره:
                return عضو
        return None
    
    def _پیدا_کن_کتاب(self, شابک):
        for کتاب in self.کتاب‌ها:
            if کتاب.شابک == شابک:
                return کتاب
        return None

💻 استفاده از پروژه

# ساخت کتابخانه
کتابخانه_من = کتابخانه("کتابخانه مرکزی")

# اضافه کردن کتاب‌ها
کتاب1 = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز", 1967, "978-123")
کتاب2 = کتاب("بوف کور", "صادق هدایت", 1937, "978-456")
کتاب3 = کتاب("شازده کوچولو", "آنتوان دوسنت‌اگزوپری", 1943, "978-789")

کتابخانه_من.اضافه_کتاب(کتاب1)
کتابخانه_من.اضافه_کتاب(کتاب2)
کتابخانه_من.اضافه_کتاب(کتاب3)

# ثبت اعضا
عضو1 = عضو("علی رضایی", "M001")
عضو2 = عضو("زهرا احمدی", "M002")
کتابخانه_من.ثبت_عضو(عضو1)
کتابخانه_من.ثبت_عضو(عضو2)

# نمایش کتاب‌ها
print(کتابخانه_من.نمایش_کتاب‌ها())

# امانت کتاب
print(کتابخانه_من.امانت_کتاب("M001", "978-123"))

# نمایش کتاب‌ها بعد از امانت
print(کتابخانه_من.نمایش_کتاب‌ها())

# بازگرداندن کتاب
print(کتابخانه_من.بازگرداندن_کتاب("M001", "978-123"))

# جستجو
نتایج = کتابخانه_من.جستجو_کتاب("صد")
for کتاب in نتایج:
    print(f"یافت شد: {کتاب}")

💻 ذخیره در فایل

import json

class کتابخانه:
    # ... کدهای قبلی ...
    
    def ذخیره_در_فایل(self, نام_فایل):
        data = {
            "نام": self.نام,
            "کتاب‌ها": [
                {
                    "عنوان": ک.عنوان,
                    "نویسنده": ک.نویسنده,
                    "سال": ک.سال,
                    "شابک": ک.شابک,
                    "امانت_رفته": ک.امانت_رفته,
                    "امانت_گیرنده": ک.امانت_گیرنده.نام if ک.امانت_گیرنده else None
                }
                for ک in self.کتاب‌ها
            ],
            "اعضا": [
                {
                    "نام": ع.نام,
                    "شماره_عضو": ع.شماره_عضو
                }
                for ع in self.اعضا
            ]
        }
        with open(نام_فایل, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        return f"ذخیره در {نام_فایل} انجام شد"
