📖 JSON چیست؟

JSON (JavaScript Object Notation) یه فرمت سبک برای تبادل داده هست که قابل خواندن توسط انسان و قابل پردازش توسط کامپیوتره.

JSON توی وب‌سایت‌ها، APIها و پایگاه‌های داده خیلی استفاده میشه.

💡 یادت باشه: JSON در پایتون به دیکشنری شباهت داره.

💻 JSON در پایتون

import json

# داده‌های پایتون -> JSON
داده = {
    "نام": "علی",
    "سن": 25,
    "شهر": "تهران",
    "علاقه‌ها": ["برنامه‌نویسی", "ورزش", "موسیقی"],
    "مجرد": True
}

# تبدیل به JSON
json_string = json.dumps(داده, ensure_ascii=False, indent=2)
print("JSON:")
print(json_string)

# تبدیل JSON به دیکشنری
json_string = '{"نام": "علی", "سن": 25}'
parsed = json.loads(json_string)
print(f"پارسی: {parsed}")
print(f"نام: {parsed['نام']}")

💻 ذخیره و خواندن JSON

import json

داده = [
    {"نام": "علی", "سن": 25, "شهر": "تهران"},
    {"نام": "زهرا", "سن": 30, "شهر": "اصفهان"},
    {"نام": "رضا", "سن": 22, "شهر": "شیراز"}
]

# ذخیره در فایل
with open("people.json", "w", encoding="utf-8") as f:
    json.dump(داده, f, ensure_ascii=False, indent=2)

print("داده ذخیره شد!")

# خواندن از فایل
with open("people.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)

print("داده بارگیری شد:")
for person in loaded:
    print(f"  {person}")

💻 CSV چیست؟

CSV (Comma-Separated Values) یه فرمت ساده برای ذخیره داده‌های جدولی هست که با کاما جدا میشن.

💡 یادت باشه: CSV برای کار با اکسل و دیتابیس‌ها خیلی کاربردی‌ه.

💻 کار با CSV

import csv

# نوشتن CSV
data = [
    ["نام", "سن", "شهر"],
    ["علی", "25", "تهران"],
    ["زهرا", "30", "اصفهان"],
    ["رضا", "22", "شیراز"]
]

with open("people.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerows(data)

print("CSV ذخیره شد!")

# خواندن CSV
with open("people.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)

💻 CSV با DictWriter

نوشتن با دیکشنری:

import csv

data = [
    {"نام": "علی", "سن": 25, "شهر": "تهران"},
    {"نام": "زهرا", "سن": 30, "شهر": "اصفهان"},
    {"نام": "رضا", "سن": 22, "شهر": "شیراز"}
]

with open("people_dict.csv", "w", newline="", encoding="utf-8") as f:
    fieldnames = ["نام", "سن", "شهر"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    
    writer.writeheader()
    for row in data:
        writer.writerow(row)

# خواندن با DictReader
with open("people_dict.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(f"{row['نام']} - {row['سن']} سال - {row['شهر']}")

💻 تبدیل CSV به JSON

import csv
import json

def csv_to_json(csv_file, json_file):
    data = []
    with open(csv_file, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            data.append(row)
    
    with open(json_file, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

csv_to_json("people_dict.csv", "output.json")
print("تبدیل انجام شد!")

💻 کار با داده‌های حجیم CSV

import csv

def پردازش_csv_بزرگ(نام_فایل):
    with open(نام_فایل, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        
        # پردازش خط به خط
        for i, row in enumerate(reader):
            # فقط ۱۰ خط اول رو نشون بده
            if i < 10:
                print(row)
            elif i == 10:
                print("...")
                break

💡 نکته‌ی هوشمندانه

برای داده‌های حجیم، از پردازش خط به خط استفاده کن تا حافظه کمتری مصرف بشه:

import csv

# جایگزین json.load() برای فایل‌های بزرگ
def خط_به_خط_json(نام_فایل):
    with open(نام_فایل, "r", encoding="utf-8") as f:
        for line in f:
            yield json.loads(line)

# استفاده
for data in خط_به_خط_json("big.json"):
    print(data)

✍️ تمرین‌ها

تمرین ۱ (ساده): یک لیست از دیکشنری‌ها رو به JSON تبدیل کن و توی فایل ذخیره کن.

تمرین ۲ (متوسط): یک فایل CSV با اطلاعات افراد بساز و با DictReader بخون.

تمرین ۳ (چالش‌برانگیز): یک فایل CSV رو بخون و به JSON تبدیل کن، سپس دوباره JSON رو به CSV برگردون.
