🎯 هدف این درس

با Matplotlib آشنا بشی، کتابخانه‌ای برای نمودار و مصورسازی داده‌ها.

📖 Matplotlib چیست؟

Matplotlib کتابخانه‌ای برای رسم نمودار هست که:

  • نمودارهای خطی، میله‌ای، دایره‌ای، پراکندگی و ...
  • کنترل کامل بر ظاهر
  • قابل تعامل و ذخیره
  • سازگاری با NumPy و Pandas

💡 نصب: pip install matplotlib

💻 نمودار خطی

import matplotlib.pyplot as plt
import numpy as np

# داده‌ها
x = [1, 2, 3, 4, 5]
y = [2, 4, 6, 8, 10]

# رسم نمودار
plt.plot(x, y)
plt.title("نمودار خطی")
plt.xlabel("محور X")
plt.ylabel("محور Y")
plt.grid(True)
plt.show()

# با داده‌های بیشتر
x = np.linspace(0, 10, 100)
y = np.sin(x)

plt.plot(x, y, label="سینوس", color="red", linestyle="--", linewidth=2)
plt.plot(x, np.cos(x), label="کسینوس", color="blue", linestyle="-", linewidth=2)
plt.legend()
plt.grid(True)
plt.show()

💻 نمودار میله‌ای

import matplotlib.pyplot as plt

# داده‌ها
محصولات = ["لپ‌تاپ", "موبایل", "تبلت", "هدفون"]
فروش = [100, 200, 150, 300]

# نمودار میله‌ای
plt.bar(محصولات, فروش, color=["blue", "green", "red", "purple"])
plt.title("فروش محصولات")
plt.xlabel("محصول")
plt.ylabel("فروش")
plt.show()

# نمودار میله‌ای افقی
plt.barh(محصولات, فروش, color="orange")
plt.title("فروش محصولات (افقی)")
plt.xlabel("فروش")
plt.show()

💻 نمودار دایره‌ای

import matplotlib.pyplot as plt

# داده‌ها
برچسب‌ها = ["پایتون", "جاوا", "C++", "JavaScript", "دیگر"]
سهم = [30, 25, 20, 15, 10]

# نمودار دایره‌ای
plt.pie(سهم, labels=برچسب‌ها, autopct="%1.1f%%", startangle=90)
plt.title("محبوبیت زبان‌های برنامه‌نویسی")
plt.axis("equal")  # دایره کامل
plt.show()

💻 نمودار پراکندگی

import matplotlib.pyplot as plt
import numpy as np

# داده‌های تصادفی
np.random.seed(42)
x = np.random.randn(100)
y = np.random.randn(100)
colors = np.random.rand(100)
sizes = np.random.randint(20, 200, 100)

# نمودار پراکندگی
plt.scatter(x, y, c=colors, s=sizes, alpha=0.5)
plt.title("نمودار پراکندگی")
plt.xlabel("X")
plt.ylabel("Y")
plt.colorbar()
plt.show()

💻 چندین نمودار در یک شکل

import matplotlib.pyplot as plt
import numpy as np

# ایجاد شکل با ۲×۲ نمودار
fig, axes = plt.subplots(2, 2, figsize=(10, 8))

# نمودار اول
axes[0, 0].plot([1, 2, 3, 4], [1, 4, 9, 16])
axes[0, 0].set_title("خطی")

# نمودار دوم
axes[0, 1].bar(["A", "B", "C"], [10, 20, 15])
axes[0, 1].set_title("میله‌ای")

# نمودار سوم
axes[1, 0].pie([25, 35, 40], labels=["x", "y", "z"])
axes[1, 0].set_title("دایره‌ای")

# نمودار چهارم
axes[1, 1].scatter(np.random.randn(50), np.random.randn(50))
axes[1, 1].set_title("پراکندگی")

plt.tight_layout()
plt.show()

💻 سفارشی‌سازی

import matplotlib.pyplot as plt
import numpy as np

plt.style.use("seaborn-v0_8")  # استایل زیبا

# داده‌ها
x = np.linspace(0, 10, 100)
y1 = np.sin(x)
y2 = np.cos(x)

# رسم با تنظیمات
plt.figure(figsize=(10, 6))
plt.plot(x, y1, label="سینوس", color="#FF6B6B", linestyle="-", linewidth=2, marker="o", markersize=4)
plt.plot(x, y2, label="کسینوس", color="#4ECDC4", linestyle="--", linewidth=2, marker="s", markersize=4)

plt.title("توابع مثلثاتی", fontsize=16, fontweight="bold")
plt.xlabel("x", fontsize=14)
plt.ylabel("y", fontsize=14)
plt.legend(loc="upper right", fontsize=12)
plt.grid(True, alpha=0.3)
plt.axhline(y=0, color="black", linestyle="-", linewidth=0.5)
plt.axvline(x=0, color="black", linestyle="-", linewidth=0.5)
plt.tight_layout()
plt.show()

💡 نکته‌ی هوشمندانه

از subplots با Pandas برای رسم سریع استفاده کن:

import pandas as pd
import matplotlib.pyplot as plt

df = pd.DataFrame({
    "سال": [2018, 2019, 2020, 2021, 2022],
    "فروش": [100, 120, 90, 150, 180],
    "سود": [20, 25, 18, 30, 35]
})

fig, axes = plt.subplots(1, 2, figsize=(12, 5))
df.plot(x="سال", y="فروش", ax=axes[0], kind="line", marker="o")
df.plot(x="سال", y="سود", ax=axes[1], kind="bar", color="green")
plt.tight_layout()
plt.show()
