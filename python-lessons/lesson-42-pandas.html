با Pandas آشنا بشی، کتابخانه‌ای قدرتمند برای تحلیل و دستکاری داده‌ها.

📖 Pandas چیست؟

Pandas کتابخانه‌ای برای تحلیل داده هست که:

  • ساختارهای داده‌ی Series و DataFrame
  • خواندن/نوشتن فرمت‌های مختلف (CSV, Excel, JSON, SQL)
  • پاکسازی و آماده‌سازی داده
  • گروه‌بندی و تجمیع

💡 نصب: pip install pandas

💻 Series و DataFrame

import pandas as pd

# Series - ستون یک‌بعدی
s = pd.Series([10, 20, 30, 40], index=["a", "b", "c", "d"])
print("Series:\n", s)

# DataFrame - جدول دو‌بعدی
data = {
    "نام": ["علی", "زهرا", "رضا", "سارا"],
    "سن": [25, 30, 22, 28],
    "شهر": ["تهران", "اصفهان", "شیراز", "مشهد"]
}
df = pd.DataFrame(data)
print("\nDataFrame:\n", df)

# دسترسی به ستون
print("\nستون سن:\n", df["سن"])
print("\nستون نام:\n", df.نام)

# دسترسی به سطر
print("\nسطر اول:\n", df.iloc[0])
print("\nسطر با ایندکس:\n", df.loc[2])

💻 فیلتر کردن داده

import pandas as pd

data = {
    "نام": ["علی", "زهرا", "رضا", "سارا", "مهدی"],
    "سن": [25, 30, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", "تهران", "مشهد", "شیراز"],
    "حقوق": [5000, 6000, 4500, 7000, 8000]
}
df = pd.DataFrame(data)

# شرط ساده
print("سن > 25:\n", df[df["سن"] > 25])

# شرط چندگانه
print("\nسن > 25 و حقوق > 6000:\n", df[(df["سن"] > 25) & (df["حقوق"] > 6000)])

# عضو بودن در لیست
print("\nشهرهای تهران یا شیراز:\n", df[df["شهر"].isin(["تهران", "شیراز"])])

# جستجو در متن
print("\nنام‌های با 'ر':\n", df[df["نام"].str.contains("ر")])

# انتخاب ستون‌ها
print("\nفقط نام و حقوق:\n", df[["نام", "حقوق"]])

💻 گروه‌بندی و تجمیع

import pandas as pd

data = {
    "محصول": ["لپ‌تاپ", "موبایل", "لپ‌تاپ", "تبلت", "موبایل", "لپ‌تاپ"],
    "فروش": [1000, 500, 1200, 300, 600, 900],
    "تعداد": [5, 10, 4, 8, 12, 6]
}
df = pd.DataFrame(data)

# گروه‌بندی
grouped = df.groupby("محصول")
print("گروه‌بندی:\n", grouped)

# مجموع
print("\nمجموع فروش:\n", grouped["فروش"].sum())

# میانگین
print("\nمیانگین فروش:\n", grouped["فروش"].mean())

# آمار کامل
print("\nآمار فروش:\n", grouped["فروش"].agg(["sum", "mean", "max", "min", "count"]))

# چندین ستون
print("\nچندین ستون:\n", grouped.agg({"فروش": "sum", "تعداد": "mean"}))

💻 مرتب‌سازی و تغییر

import pandas as pd

data = {
    "نام": ["علی", "زهرا", "رضا", "سارا", "مهدی"],
    "سن": [25, 30, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", "تهران", "مشهد", "شیراز"]
}
df = pd.DataFrame(data)

# مرتب‌سازی بر اساس سن
print("مرتب‌سازی بر اساس سن:\n", df.sort_values("سن"))

# مرتب‌سازی نزولی
print("\nمرتب‌سازی نزولی:\n", df.sort_values("سن", ascending=False))

# اضافه کردن ستون جدید
df["سن_افزایش"] = df["سن"] + 5
print("\nستون جدید:\n", df)

# تغییر نام ستون
df = df.rename(columns={"شهر": "استان"})
print("\nتغییر نام:\n", df)

# حذف ستون
df = df.drop("سن_افزایش", axis=1)
print("\nحذف ستون:\n", df)

💻 مقادیر گمشده

import pandas as pd
import numpy as np

data = {
    "نام": ["علی", "زهرا", "رضا", None, "مهدی"],
    "سن": [25, np.nan, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", None, "مشهد", "شیراز"]
}
df = pd.DataFrame(data)

print("داده:\n", df)

# بررسی مقادیر گمشده
print("\nمقادیر گمشده:\n", df.isnull())

# تعداد گمشده‌ها
print("\nتعداد گمشده‌ها:\n", df.isnull().sum())

# حذف سطرهای گمشده
print("\nحذف گمشده:\n", df.dropna())

# پر کردن گمشده‌ها
print("\nپر کردن با ۰:\n", df.fillna(0))

# پر کردن با مقدار خاص
df["سن"] = df["سن"].fillna(df["سن"].mean())
df["شهر"] = df["شهر"].fillna("نامشخص")
print("\nپر کردن با میانگین:\n", df)

💻 ذخیره داده

import pandas as pd

df = pd.DataFrame({
    "نام": ["علی", "زهرا", "رضا"],
    "سن": [25, 30, 22],
    "شهر": ["تهران", "اصفهان", "شیراز"]
})

# ذخیره به CSV
df.to_csv("output.csv", index=False, encoding="utf-8")

# ذخیره به Excel
df.to_excel("output.xlsx", index=False)

# ذخیره به JSON
df.to_json("output.json", orient="records", force_ascii=False)

# ذخیره به Clipboard
df.to_clipboard()

print("فایل‌ها ذخیره شدند!")

💡 نکته‌ی هوشمندانه

از pandas با matplotlib برای نمودار استفاده کن:

import pandas as pd
import matplotlib.pyplot as plt

df = pd.DataFrame({
    "محصول": ["A", "B", "C", "D"],
    "فروش": [100, 200, 150, 300]
})

# نمودار ساده
df.plot(x="محصول", y="فروش", kind="bar")
plt.title("فروش محصولات")
plt.show()

# نمودار خطی
df.plot(x="محصول", y="فروش", kind="line")
plt.title("روند فروش")
plt.show()

✍️ تمرین‌ها

تمرین ۱: یک DataFrame با اطلاعات ۵ نفر بساز (نام، سن، شهر، حقوق) و آمار پایه رو نمایش بده.

تمرین ۲: DataFrame رو بر اساس سن مرتب کن و فقط افرادی که حقوق > ۵۰۰۰ هستن رو نمایش بده.

تمرین ۳: DataFrame رو بر اساس شهر گروه‌بندی کن و میانگین حقوق هر شهر رو حساب کن.
