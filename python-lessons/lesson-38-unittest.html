🎯 هدف این درس

توی این درس با تست‌نویسی آشنا میشی که بهت کمک می‌کنه کدهای با کیفیت‌تر و بدون اشکال بنویسی.

📖 تست‌نویسی چیست؟

تست‌نویسی یعنی نوشتن کدهایی که کد اصلی رو تست می‌کنن و مطمئن میشن درست کار می‌کنه.

  • پیدا کردن باگ‌ها زودتر
  • اطمینان از کارکرد صحیح
  • تغییر کد با خیال راحت

💡 یادت باشه: تست‌نویسی یکی از نشانه‌های برنامه‌نویس حرفه‌ای هست.

💻 ماژول unittest

import unittest

# کد اصلی
def جمع(a, b):
    return a + b

def تفریق(a, b):
    return a - b

# کلاس تست
class Testحساب(unittest.TestCase):
    def test_جمع(self):
        self.assertEqual(جمع(5, 3), 8)
        self.assertEqual(جمع(-1, 1), 0)
        self.assertEqual(جمع(0, 0), 0)
    
    def test_تفریق(self):
        self.assertEqual(تفریق(10, 5), 5)
        self.assertEqual(تفریق(0, 5), -5)
        self.assertEqual(تفریق(-5, -5), 0)

# اجرا
if __name__ == "__main__":
    unittest.main()

💻 تست با setUp

import unittest

class TestList(unittest.TestCase):
    def setUp(self):
        # قبل از هر تست اجرا میشه
        self.data = [1, 2, 3, 4, 5]
        print("setUp اجرا شد")
    
    def tearDown(self):
        # بعد از هر تست اجرا میشه
        print("tearDown اجرا شد")
    
    def test_len(self):
        self.assertEqual(len(self.data), 5)
    
    def test_sum(self):
        self.assertEqual(sum(self.data), 15)
    
    def test_max(self):
        self.assertEqual(max(self.data), 5)

if __name__ == "__main__":
    unittest.main()

💻 انواع Assertions

import unittest

class TestAssertions(unittest.TestCase):
    def test_assertions(self):
        # برابری
        self.assertEqual(5 + 3, 8)
        
        # نابرابری
        self.assertNotEqual(5 + 3, 9)
        
        # درستی
        self.assertTrue(5 > 3)
        self.assertTrue("hello".isalpha())
        
        # نادرستی
        self.assertFalse(5 < 3)
        self.assertFalse("123".isalpha())
        
        # None
        self.assertIsNone(None)
        self.assertIsNotNone("hello")
        
        # عضویت
        self.assertIn(3, [1, 2, 3, 4])
        self.assertNotIn(5, [1, 2, 3, 4])
        
        # استثناها
        with self.assertRaises(ZeroDivisionError):
            result = 10 / 0

💻 تست تابع با خطا

import unittest

def تقسیم(a, b):
    if b == 0:
        raise ValueError("تقسیم بر صفر!")
    return a / b

class Testتقسیم(unittest.TestCase):
    def test_تقسیم_معمولی(self):
        self.assertEqual(تقسیم(10, 2), 5)
        self.assertEqual(تقسیم(7, 2), 3.5)
    
    def test_تقسیم_بر_صفر(self):
        with self.assertRaises(ValueError):
            تقسیم(10, 0)
    
    def test_تقسیم_با_عدد_منفی(self):
        self.assertEqual(تقسیم(-10, 2), -5)
        self.assertEqual(تقسیم(10, -2), -5)

if __name__ == "__main__":
    unittest.main()

💻 تست کلاس

import unittest

class حساب:
    def __init__(self, موجودی=0):
        self.موجودی = موجودی
    
    def واریز(self, مبلغ):
        if مبلغ <= 0:
            raise ValueError("مبلغ باید مثبت باشد!")
        self.موجودی += مبلغ
        return self.موجودی
    
    def برداشت(self, مبلغ):
        if مبلغ <= 0:
            raise ValueError("مبلغ باید مثبت باشد!")
        if مبلغ > self.موجودی:
            raise ValueError("موجودی کافی نیست!")
        self.موجودی -= مبلغ
        return self.موجودی

class Testحساب(unittest.TestCase):
    def setUp(self):
        self.حساب = حساب(1000)
    
    def test_واریز(self):
        self.assertEqual(self.حساب.واریز(500), 1500)
        self.assertEqual(self.حساب.واریز(200), 1700)
    
    def test_برداشت(self):
        self.assertEqual(self.حساب.برداشت(300), 700)
        self.assertEqual(self.حساب.برداشت(100), 600)
    
    def test_واریز_با_مبلغ_منفی(self):
        with self.assertRaises(ValueError):
            self.حساب.واریز(-100)
    
    def test_برداشت_بیشتر_از_موجودی(self):
        with self.assertRaises(ValueError):
            self.حساب.برداشت(2000)

if __name__ == "__main__":
    unittest.main()

💻 تست با mock

import unittest
from unittest.mock import Mock, patch

class کاربر:
    def __init__(self, نام):
        self.نام = نام
    
    def سلام_بگو(self):
        return f"سلام {self.نام}!"

class Testکاربر(unittest.TestCase):
    def test_سلام_بگو(self):
        # ساخت mock
        mock_user = Mock()
        mock_user.نام = "علی"
        mock_user.سلام_بگو.return_value = "سلام علی!"
        
        self.assertEqual(mock_user.سلام_بگو(), "سلام علی!")
        mock_user.سلام_بگو.assert_called_once()
    
    @patch('builtins.print')
    def test_چاپ_با_patch(self, mock_print):
        def چاپ_کن(پیام):
            print(پیام)
        
        چاپ_کن("سلام")
        mock_print.assert_called_with("سلام")

if __name__ == "__main__":
    unittest.main()

💡 نکته‌ی هوشمندانه

از Test-Driven Development (TDD) استفاده کن:

  1. اول تست رو بنویس
  2. تست شکست می‌خوره (کد هنوز نوشته نشده)
  3. کد رو بنویس تا تست پاس بشه
  4. رفکتور کن و دوباره تست کن
# مرحله ۱: تست بنویس
def test_جمع():
    assert جمع(2, 3) == 5  # این خطا میده

# مرحله ۳: کد بنویس
def جمع(a, b):
    return a + b

✍️ تمرین‌ها

تمرین ۱ (ساده): برای تابع ضرب یک تست بنویس.

تمرین ۲ (متوسط): برای کلاس مستطیل (مساحت و محیط) تست بنویس.

تمرین ۳ (چالش‌برانگیز): برای کلاس کتابخانه (که در درس ۳۲ ساختیم) تست بنویس.
