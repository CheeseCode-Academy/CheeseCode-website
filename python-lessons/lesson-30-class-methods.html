🔧 درس ۳۰: متدهای کلاس و استاتیک

سطح: متوسط | زمان مطالعه: ۱۸ دقیقه

🎯 هدف این درس

توی این درس با متدهای کلاس و متدهای استاتیک آشنا میشی و تفاوت‌شون رو با متدهای معمولی یاد می‌گیری.

📖 انواع متدها در کلاس

  • متد معمولی - با self کار می‌کنه (روی شیء)
  • متد کلاس - با cls کار می‌کنه (روی کلاس) - @classmethod
  • متد استاتیک - به کلاس یا شیء وابسته نیست - @staticmethod

💡 یادت باشه: متدهای کلاس و استاتیک رو می‌تونی بدون ساختن شیء هم صدا بزنی.

💻 متد معمولی

class شخص:
    def __init__(self, نام):
        self.نام = نام
    
    def معرفی(self):  # متد معمولی
        return f"سلام! من {self.نام} هستم"

علی = شخص("علی")
print(علی.معرفی())  # فقط با شیء قابل صدا زدنه

💻 متد کلاس (@classmethod)

class دانشجو:
    دانشگاه = "دانشگاه تهران"
    
    def __init__(self, نام):
        self.نام = نام
    
    @classmethod
    def تغییر_دانشگاه(cls, دانشگاه_جدید):
        cls.دانشگاه = دانشگاه_جدید
    
    @classmethod
    def از_رشته(cls, نام, رشته):
        # سازنده جایگزین
        دانشجو = cls(نام)
        دانشجو.رشته = رشته
        return دانشجو

# استفاده بدون شیء
print(دانشجو.دانشگاه)  # دانشگاه تهران
دانشجو.تغییر_دانشگاه("دانشگاه صنعتی")
print(دانشجو.دانشگاه)  # دانشگاه صنعتی

# سازنده جایگزین
علی = دانشجو.از_رشته("علی", "مهندسی")
print(علی.نام)  # علی
print(علی.رشته)  # مهندسی

💻 متد استاتیک (@staticmethod)

class ریاضی:
    @staticmethod
    def جمع(a, b):
        return a + b
    
    @staticmethod
    def ضرب(a, b):
        return a * b
    
    @staticmethod
    def اول_است(n):
        if n <= 1:
            return False
        for i in range(2, int(n ** 0.5) + 1):
            if n % i == 0:
                return False
        return True

# استفاده بدون شیء
print(ریاضی.جمع(5, 3))  # 8
print(ریاضی.ضرب(4, 2))  # 8
print(ریاضی.اول_است(7))  # True
print(ریاضی.اول_است(10))  # False

💻 تفاوت‌های کلیدی

class مثال:
    متغیر_کلاس = "مقدار کلاس"
    
    def __init__(self, نام):
        self.نام = نام
    
    def متد_معمولی(self):
        return f"self: {self.نام}"
    
    @classmethod
    def متد_کلاس(cls):
        return f"cls: {cls.متغیر_کلاس}"
    
    @staticmethod
    def متد_استاتیک():
        return "متد استاتیک"

# متد معمولی - به شیء نیاز داره
م = مثال("علی")
print(م.متد_معمولی())  # self: علی

# متد کلاس - به کلاس یا شیء وصل میشه
print(مثال.متد_کلاس())  # cls: مقدار کلاس
print(م.متد_کلاس())     # cls: مقدار کلاس

# متد استاتیک - مستقل
print(مثال.متد_استاتیک())  # متد استاتیک
print(م.متد_استاتیک())     # متد استاتیک

💻 کاربردهای عملی

class تاریخ:
    def __init__(self, روز, ماه, سال):
        self.روز = روز
        self.ماه = ماه
        self.سال = سال
    
    @classmethod
    def از_رشته(cls, تاریخ_رشته):
        # "2026-07-21" -> تاریخ(2026, 7, 21)
        سال, ماه, روز = map(int, تاریخ_رشته.split("-"))
        return cls(روز, ماه, سال)
    
    @staticmethod
    def معتبر_است(روز, ماه, سال):
        # بررسی ساده
        if ماه < 1 or ماه > 12:
            return False
        if روز < 1 or روز > 31:
            return False
        return True
    
    def __str__(self):
        return f"{self.سال}/{self.ماه}/{self.روز}"

# استفاده از سازنده جایگزین
تاریخ1 = تاریخ.از_رشته("2026-07-21")
print(تاریخ1)  # 2026/7/21

# استفاده از متد استاتیک
print(تاریخ.معتبر_است(31, 12, 2026))  # True
print(تاریخ.معتبر_است(32, 12, 2026))  # False

💻 متد کلاس برای inheritance

class حیوان:
    نوع = "حیوان"
    
    @classmethod
    def اطلاعات(cls):
        return f"من یک {cls.نوع} هستم"

class سگ(حیوان):
    نوع = "سگ"

class گربه(حیوان):
    نوع = "گربه"

print(حیوان.اطلاعات())  # من یک حیوان هستم
print(سگ.اطلاعات())     # من یک سگ هستم
print(گربه.اطلاعات())   # من یک گربه هستم

💻 استفاده عملی از @staticmethod

class ابزار:
    @staticmethod
    def پاک_کن_فاصله(متن):
        return " ".join(متن.split())
    
    @staticmethod
    def معکوس_کن(متن):
        return متن[::-1]
    
    @staticmethod
    def بزرگ_کن(متن):
        return متن.upper()

print(ابزار.پاک_کن_فاصله("  سلام   علی  "))  # سلام علی
print(ابزار.معکوس_کن("سلام"))  # ملس
print(ابزار.بزرگ_کن("hello"))  # HELLO

🧠 اشتباهات رایج

❌ اشتباه: استفاده از self در متد کلاس

✅ راه‌حل: در متد کلاس از cls استفاده کن.

❌ اشتباه: استفاده از cls در متد استاتیک

✅ راه‌حل: متد استاتیک به کلاس وابسته نیست، از هیچی استفاده نکن.

❌ اشتباه: فراموش کردن دکوریتورها: @classmethod یا @staticmethod

✅ راه‌حل: حتماً دکوریتور رو قبل از متد بذار.

💡 نکته‌ی هوشمندانه

از @classmethod برای سازنده‌های جایگزین استفاده کن که ورودی‌های مختلفی رو قبول کنن:

class نقطه:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    @classmethod
    def از_قطبی(cls, فاصله, زاویه):
        import math
        return cls(فاصله * math.cos(زاویه), فاصله * math.sin(زاویه))
    
    def __str__(self):
        return f"({self.x}, {self.y})"

نقطه1 = نقطه(3, 4)
نقطه2 = نقطه.از_قطبی(5, 0.927)
print(نقطه1)  # (3, 4)
print(نقطه2)  # (~3, ~4)

✍️ تمرین‌ها

تمرین ۱ (ساده): کلاسی با یه متد کلاس بساز که یه پیام چاپ کنه.

تمرین ۲ (متوسط): کلاس تبدیل بساز با متدهای استاتیک برای تبدیل دما (سلسیوس به فارنهایت و بالعکس).

تمرین ۳ (چالش‌برانگیز): کلاس کارخانه بساز که با متد کلاس، اشیاء مختلف تولید کنه.
