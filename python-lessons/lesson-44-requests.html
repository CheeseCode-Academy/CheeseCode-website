🎯 هدف این درس

با Requests آشنا بشی، کتابخانه‌ای برای کار با HTTP و دریافت داده از اینترنت.

📖 Requests چیست؟

Requests کتابخانه‌ای برای درخواست HTTP هست که:

  • دریافت داده از APIها
  • دانلود فایل
  • ارسال داده به سرور
  • کار با کوکی‌ها و هدرها

💡 نصب: pip install requests

💻 دریافت داده (GET)

import requests

# درخواست GET ساده
response = requests.get("https://api.github.com")

print(f"وضعیت: {response.status_code}")
print(f"متن: {response.text[:200]}...")
print(f"JSON: {response.json()}")

# با پارامترها
params = {"q": "python", "sort": "stars"}
response = requests.get("https://api.github.com/search/repositories", params=params)
print(f"URL: {response.url}")
print(f"تعداد نتایج: {response.json()['total_count']}")

💻 ارسال داده (POST)

import requests

# ارسال JSON
data = {"title": "New Post", "body": "This is a new post", "userId": 1}

print(f"وضعیت: {response.status_code}")

💻 هدرها

import requests

# هدرهای سفارشی
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
    "Accept": "application/json",
    "Authorization": "Bearer YOUR_TOKEN"
}

response = requests.get("https://api.github.com/user", headers=headers)
print(f"وضعیت: {response.status_code}")

# دیدن هدرهای پاسخ
print("هدرهای پاسخ:")
for key, value in response.headers.items():
    print(f"  {key}: {value}")

💻 خطاگیری

import requests

try:
    response = requests.get("https://api.github.com/user", timeout=5)
    response.raise_for_status()  # خطا پرتاب میکنه
    print("موفقیت!")
except requests.exceptions.Timeout:
    print("زمان پاسخ‌دهی تمام شد!")
except requests.exceptions.ConnectionError:
    print("خطا در اتصال!")
except requests.exceptions.HTTPError as e:
    print(f"خطای HTTP: {e}")
except requests.exceptions.RequestException as e:
    print(f"خطا: {e}")

💻 دانلود فایل

import requests

# دانلود تصویر
url = "https://www.python.org/static/img/python-logo.png"
response = requests.get(url, stream=True)

if response.status_code == 200:
    with open("python-logo.png", "wb") as f:
        for chunk in response.iter_content(chunk_size=8192):
            f.write(chunk)
    print("تصویر دانلود شد!")

# دانلود با پیشرفت
url = "https://speedtest.tele2.net/10MB.zip"
response = requests.get(url, stream=True)
total_size = int(response.headers.get("content-length", 0))
downloaded = 0

with open("file.zip", "wb") as f:
    for chunk in response.iter_content(chunk_size=8192):
        f.write(chunk)
        downloaded += len(chunk)
        progress = (downloaded / total_size) * 100
        print(f"\rپیشرفت: {progress:.1f}%", end="")
print("\nدانلود کامل شد!")

💻 کار با API (مثال)

import requests

class GitHubAPI:
    def __init__(self, token=None):
        self.base_url = "https://api.github.com"
        self.headers = {}
        if token:
            self.headers["Authorization"] = f"Bearer {token}"
    
    def get_user(self, username):
        url = f"{self.base_url}/users/{username}"
        response = requests.get(url, headers=self.headers)
        if response.status_code == 200:
            return response.json()
        return None
    
    def get_repos(self, username):
        url = f"{self.base_url}/users/{username}/repos"
        response = requests.get(url, headers=self.headers)
        if response.status_code == 200:
            return response.json()
        return None

# استفاده
api = GitHubAPI()
user = api.get_user("python")
if user:
    print(f"نام: {user['name']}")
    print(f"دنبال‌کنندگان: {user['followers']}")
    print(f"دنبال‌شوندگان: {user['following']}")
    print(f"مخزن‌ها: {user['public_repos']}")

repos = api.get_repos("python")
if repos:
    print(f"\n۵ مخزن اول:")
    for repo in repos[:5]:
        print(f"  - {repo['name']}: ★{repo['stargazers_count']}")

💻 جلسه (Session)

import requests

# ایجاد جلسه برای نگهداری کوکی‌ها
session = requests.Session()
session.headers.update({"User-Agent": "MyApp/1.0"})

# لاگین
login_data = {"username": "user", "password": "pass"}
response = session.post("https://example.com/login", data=login_data)

# درخواست‌های بعدی با جلسه
response = session.get("https://example.com/dashboard")
print(response.text[:100])

# بستن جلسه
session.close()

💡 نکته‌ی هوشمندانه

از Retry برای درخواست‌های ناموفق استفاده کن:

from requests.adapters import HTTPAdapter
from requests.packages.urllib3.util.retry import Retry

session = requests.Session()
retry = Retry(total=3, backoff_factor=1, status_forcelist=[500, 502, 503, 504])
adapter = HTTPAdapter(max_retries=retry)
session.mount("http://", adapter)
session.mount("https://", adapter)

response = session.get("https://api.example.com/data")
print(response.status_code)
