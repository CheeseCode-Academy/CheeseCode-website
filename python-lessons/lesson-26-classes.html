📖 کلاس و شی چیست؟

کلاس (Class) مثل یه قالب یا نقشه هست که مشخص می‌کنه یه شیء چه ویژگی‌ها و چه کارهایی داره.

شیء (Object) یه نمونه (Instance) از کلاس هست. مثل یه ماشین که از روی نقشه ساخته شده.

💡 یادت باشه: کلاس = نقشه، شیء = چیزی که از روی نقشه ساخته میشه.

💻 تعریف کلاس ساده

class شخص:
    def __init__(self, نام, سن):
        self.نام = نام
        self.سن = سن
    
    def معرفی(self):
        return f"سلام! من {self.نام} هستم و {self.سن} سال دارم."

# ایجاد شیء
علی = شخص("علی", 25)
print(علی.معرفی())

زهرا = شخص("زهرا", 30)
print(زهرا.معرفی())
خروجی: 
سلام! من علی هستم و 25 سال دارم.
سلام! من زهرا هستم و 30 سال دارم.

💡 نکته: __init__ متد سازنده (Constructor) هست که موقع ساخت شیء فراخوانی میشه.

💻 متدهای کلاس

class ماشین:
    def __init__(self, نام, سرعت):
        self.نام = نام
        self.سرعت = سرعت
    
    def شتاب(self, مقدار):
        self.سرعت += مقدار
        return f"سرعت جدید: {self.سرعت}"
    
    def ترمز(self, مقدار):
        self.سرعت -= مقدار
        if self.سرعت < 0:
            self.سرعت = 0
        return f"سرعت جدید: {self.سرعت}"

پژو = ماشین("پژو", 0)
print(پژو.شتاب(20))
print(پژو.شتاب(30))
print(پژو.ترمز(10))
print(پژو.ترمز(50))
خروجی: 
سرعت جدید: 20
سرعت جدید: 50
سرعت جدید: 40
سرعت جدید: 0

💻 متغیرهای کلاس و شیء

class دانشجو:
    دانشگاه = "دانشگاه تهران"  # متغیر کلاس
    
    def __init__(self, نام):
        self.نام = نام  # متغیر شیء
        self.شماره = 0  # متغیر شیء

علی = دانشجو("علی")
زهرا = دانشجو("زهرا")

print(علی.دانشگاه)  # دانشگاه تهران
print(زهرا.دانشگاه)  # دانشگاه تهران

# تغییر متغیر کلاس
دانشجو.دانشگاه = "دانشگاه صنعتی"

print(علی.دانشگاه)  # دانشگاه صنعتی
print(زهرا.دانشگاه)  # دانشگاه صنعتی

💻 متدهای کلاس و استاتیک

class حساب:
    نرخ_بهره = 0.05
    
    def __init__(self, موجودی):
        self.موجودی = موجودی
    
    def سود_سالانه(self):
        return self.موجودی * حساب.نرخ_بهره
    
    @classmethod
    def تغییر_نرخ(cls, نرخ):
        cls.نرخ_بهره = نرخ
    
    @staticmethod
    def اعتبار_سنجی(مبلغ):
        return مبلغ > 0

حساب.تغییر_نرخ(0.07)
print(حساب.نرخ_بهره)  # 0.07
print(حساب.اعتبار_سنجی(100))  # True
print(حساب.اعتبار_سنجی(-10))  # False

💻 properties (پروپرتی‌ها)

class شخص:
    def __init__(self, نام):
        self._نام = نام
    
    @property
    def نام(self):
        return self._نام
    
    @نام.setter
    def نام(self, مقدار):
        if not مقدار:
            raise ValueError("نام نمی‌تواند خالی باشد!")
        self._نام = مقدار

علی = شخص("علی")
print(علی.نام)  # علی

علی.نام = "علی رضایی"
print(علی.نام)  # علی رضایی

# علی.نام = ""  # ValueError!

💻 __str__ و __repr__

class کتاب:
    def __init__(self, عنوان, نویسنده):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
    
    def __str__(self):
        return f"📚 {self.عنوان} - {self.نویسنده}"
    
    def __repr__(self):
        return f"کتاب('{self.عنوان}', '{self.نویسنده}')"

کتاب۱ = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز")
print(کتاب۱)  # 📚 صد سال تنهایی - گابریل گارسیا مارکز
print(repr(کتاب۱))  # کتاب('صد سال تنهایی', 'گابریل گارسیا مارکز')

🧠 اشتباهات رایج

❌ اشتباه: فراموش کردن self در متدها: def معرفی(نام):

✅ راه‌حل: اولین پارامتر متدها همیشه self هست.

❌ اشتباه: فراموش کردن () موقع ایجاد شیء: علی = شخص

✅ راه‌حل: حتماً پرانتز بذار: علی = شخص()

❌ اشتباه: استفاده از متغیر کلاس با self

✅ راه‌حل: برای متغیرهای کلاس از اسم کلاس استفاده کن: کلاس.متغیر

💡 نکته‌ی هوشمندانه

از __slots__ برای بهینه‌سازی حافظه استفاده کن:

class شخص:
    __slots__ = ["نام", "سن"]
    def __init__(self, نام, سن):
        self.نام = نام
        self.سن = سن

علی = شخص("علی", 25)
# علی.شهر = "تهران"  # ارور! فقط نام و سن مجاز هستن

✍️ تمرین‌ها

تمرین ۱ (ساده): کلاسی به اسم مستطیل بساز که طول و عرض رو بگیره و مساحت و محیط رو حساب کنه.

تمرین ۲ (متوسط): کلاسی به اسم دانشجو بساز با ویژگی‌های نام، شماره دانشجویی و نمرات. متدی برای محاسبه معدل اضافه کن.

تمرین ۳ (چالش‌برانگیز): کلاسی به اسم بانک بساز که حساب‌های کاربری رو مدیریت کنه (واریز، برداشت، نمایش موجودی).
