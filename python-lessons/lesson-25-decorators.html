🎯 هدف این درس

توی این درس با دکوریتورها آشنا میشی. دکوریتورها روشی قدرتمند برای تغییر رفتار توابع بدون تغییر کد خودشون هستن.

📖 دکوریتور چیست؟

دکوریتور (Decorator) یه تابع هست که تابع دیگه‌ای رو می‌گیره و تابع جدیدی برمی‌گردونه که رفتار تابع اصلی رو تغییر میده.

فرض کن دکوریتور مثل یه لفاف هست که دور یه تابع پیچیده میشه و بهش قابلیت‌های جدید اضافه می‌کنه.

💡 یادت باشه: دکوریتورها با @ قبل از اسم تابع استفاده میشن.

💻 دکوریتور ساده

def زمان_سنج(func):
    import time
    def wrapper(*args, **kwargs):
        شروع = time.time()
        نتیجه = func(*args, **kwargs)
        پایان = time.time()
        print(f"زمان اجرا: {پایان - شروع:.4f} ثانیه")
        return نتیجه
    return wrapper

@زمان_سنج
def تابع_طولانی():
    sum([i ** 2 for i in range(1000000)])

تابع_طولانی()
خروجی (نمونه): 
زمان اجرا: 0.0523 ثانیه

💻 دکوریتور با پارامتر

def تکرار(تعداد):
    def دکوراتور(func):
        def wrapper(*args, **kwargs):
            for _ in range(تعداد):
                func(*args, **kwargs)
        return wrapper
    return دکوراتور

@تکرار(3)
def سلام():
    print("سلام!")

سلام()

💻 دکوریتور لاگینگ

def لاگ(func):
    def wrapper(*args, **kwargs):
        print(f"تابع {func.__name__} فراخوانی شد")
        print(f"آرگومان‌ها: {args}, {kwargs}")
        نتیجه = func(*args, **kwargs)
        print(f"نتیجه: {نتیجه}")
        return نتیجه
    return wrapper

@لاگ
def جمع(a, b):
    return a + b

💻 دکوریتور اعتبارسنجی

def اعتبارسنجی_سن(func):
    def wrapper(سن):
        if سن < 0:
            raise ValueError("سن نمی‌تواند منفی باشد!")
        if سن > 120:
            raise ValueError("سن نمی‌تواند بیشتر از ۱۲۰ باشد!")
        return func(سن)
    return wrapper

@اعتبارسنجی_سن
def وضعیت_سن(سن):
    if سن < 18:
        return "کودک/نوجوان"
    elif سن < 60:
        return "بزرگسال"
    else:
        return "سالمند"

print(وضعیت_سن(25))
print(وضعیت_سن(-5))

💻 دکوریتور با functools.wraps

برای حفظ اطلاعات تابع اصلی:

import functools

def لاگ(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        print(f"اجرای {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

@لاگ
def مثال():
    """این یک تابع نمونه است"""
    pass

print(مثال.__name__)  # مثال
print(مثال.__doc__)   # این یک تابع نمونه است

💻 دکوریتور کلاس

def اضافه_کننده(کلاس):
    class کلاس_جدید(کلاس):
        def سلام(self):
            return "سلام از دکوریتور!"
    return کلاس_جدید

@اضافه_کننده
class شخص:
    def __init__(self, نام):
        self.نام = نام

علی = شخص("علی")
print(علی.سلام())
خروجی: 
سلام از دکوریتور!

🧠 اشتباهات رایج

❌ اشتباه: فراموش کردن return wrapper در دکوریتور

✅ راه‌حل: دکوریتور حتماً باید تابع wrapper رو برگردونه.

❌ اشتباه: فراموش کردن *args, **kwargs در wrapper

✅ راه‌حل: برای اینکه دکوریتور روی هر تابعی کار کنه، از *args, **kwargs استفاده کن.

❌ اشتباه: فراموش کردن functools.wraps

✅ راه‌حل: از @functools.wraps(func) برای حفظ اطلاعات تابع اصلی استفاده کن.

💡 نکته‌ی هوشمندانه

از دکوریتور برای ذخیره‌سازی در cache استفاده کن:

def cache(func):
    ذخیره = {}
    def wrapper(*args):
        if args in ذخیره:
            print("از کش استفاده شد!")
            return ذخیره[args]
        نتیجه = func(*args)
        ذخیره[args] = نتیجه
        return نتیجه
    return wrapper

@cache
def فیبوناچی(n):
    if n < 2:
        return n
    return فیبوناچی(n-1) + فیبوناچی(n-2)

print(فیبوناچی(35))

✍️ تمرین‌ها

تمرین ۱ (ساده): دکوریتوری بساز که قبل از هر تابع "شروع..." و بعدش "پایان..." چاپ کنه.

تمرین ۲ (متوسط): دکوریتوری بساز که خطاهای تابع رو بگیره و چاپ کنه.

تمرین ۳ (چالش‌برانگیز): دکوریتوری بساز که زمان اجرای تابع رو اندازه بگیره و گزارش بده.
