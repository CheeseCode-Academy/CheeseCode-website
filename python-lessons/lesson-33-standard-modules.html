🎯 هدف این درس

توی این درس با ماژول‌های استاندارد مهم پایتون آشنا میشی که کارهای روزمره رو خیلی راحت‌تر می‌کنن.

📖 ماژول os

کار با سیستم‌عامل:

import os

# اطلاعات سیستم
print(f"سیستم‌عامل: {os.name}")
print(f"پوشه فعلی: {os.getcwd()}")

# کار با فایل و پوشه
os.mkdir("پوشه_جدید")  # ایجاد پوشه
os.chdir("پوشه_جدید")  # تغییر پوشه
print(f"پوشه جدید: {os.getcwd()}")

# لیست فایل‌ها
print("فایل‌ها:", os.listdir("."))

# حذف پوشه
os.chdir("..")
os.rmdir("پوشه_جدید")

# مسیرها
مسیر = os.path.join("پوشه", "فایل.txt")
print(f"مسیر: {مسیر}")
print(f"وجود دارد: {os.path.exists(مسیر)}")
print(f"فایل است: {os.path.isfile(مسیر)}")
print(f"پوشه است: {os.path.isdir(مسیر)}")

💻 ماژول datetime

کار با تاریخ و زمان:

import datetime

# زمان حال

print(f"اکنون: {اکنون}")

print(f"زمان: {اکنون.time()}")

# اجزای زمان
print(f"سال: {اکنون.year}")
print(f"ماه: {اکنون.month}")
print(f"روز: {اکنون.day}")
print(f"ساعت: {اکنون.hour}")

تاریخ1 = datetime.datetime(2026, 7, 21)

print(f"تفاوت: {تفاوت}")
print(f"روزها: {تفاوت.days}")
print(f"ثانیه‌ها: {تفاوت.seconds}")

💻 ماژول random

تولید اعداد تصادفی:

import random

# عدد تصادفی
print(f"تصادفی: {random.random()}")
print(f"بین ۱ تا ۱۰: {random.randint(1, 10)}")
print(f"بین ۰ تا ۱: {random.uniform(0, 1)}")

# انتخاب تصادفی
لیست = ["سیب", "موز", "پرتقال", "انگور"]
print(f"انتخاب: {random.choice(لیست)}")
print(f"چند انتخاب: {random.choices(لیست, k=2)}")

# مرتب‌سازی تصادفی
random.shuffle(لیست)
print(f"مرتب‌سازی: {لیست}")

# نمونه‌گیری
print(f"نمونه: {random.sample(لیست, 2)}")

مرتب‌سازی: ['انگور', 'سیب', 'پرتقال', 'موز']

💻 ماژول re (عبارت‌های منظم)

پیدا کردن الگوها در متن:

import re

# جستجوی الگو
متن = "سلام! شماره تلفن من 0912-345-6789 است."
الگو = r"\d{4}-\d{3}-\d{4}"

پیدا = re.search(الگو, متن)
if پیدا:
    print(f"پیدا شد: {پیدا.group()}")

# پیدا کردن همه
شماره‌ها = re.findall(r"\d+", متن)
print(f"همه اعداد: {شماره‌ها}")

# جایگزینی
متن_جدید = re.sub(الگو, "***-***-****", متن)
print(f"متن جدید: {متن_جدید}")

# بررسی تطابق
if re.match(r"^سلام", متن):
    print("متن با سلام شروع می‌شود")

💻 ماژول collections

ساختارهای داده پیشرفته:

from collections import Counter, defaultdict, OrderedDict

# Counter - شمارش
تکرار = Counter("سلام سلام علی")
print(f"تکرار: {تکرار}")
print(f"بیشترین: {تکرار.most_common(2)}")

# defaultdict - دیکشنری با مقدار پیش‌فرض
dd = defaultdict(int)
dd["x"] += 1
dd["y"] += 2
dd["x"] += 3
print(f"defaultdict: {dict(dd)}")

# OrderedDict - دیکشنری مرتب
od = OrderedDict()
od["a"] = 1
od["b"] = 2
od["c"] = 3
print(f"ordered: {od}")
خروجی: 
تکرار: Counter({' ': 3, 'س': 2, 'ل': 2, 'ا': 2, 'م': 1, 'ع': 1, 'ی': 1})
بیشترین: [(' ', 3), ('س', 2)]
defaultdict: {'x': 4, 'y': 2}
ordered: OrderedDict([('a', 1), ('b', 2), ('c', 3)])

💻 ماژول json

کار با داده‌های JSON:

import json

# تبدیل دیکشنری به JSON
data = {
    "نام": "علی",
    "سن": 25,
    "شهر": "تهران",
    "علاقه‌ها": ["برنامه‌نویسی", "ورزش"]
}

json_string = json.dumps(data, ensure_ascii=False, indent=2)
print(f"JSON:\n{json_string}")

# تبدیل JSON به دیکشنری
parsed = json.loads(json_string)
print(f"پارسی: {parsed}")

# ذخیره در فایل
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

# خواندن از فایل
with open("data.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)
print(f"بارگیری: {loaded}")

💻 ماژول time

کار با زمان:

import time

# زمان فعلی به ثانیه
print(f"زمان: {time.time()}")

# مکث
print("صبر کنید...")
time.sleep(2)
print("ادامه!")

# زمان با فرمت
print(f"زمان محلی: {time.localtime()}")
print(f"زمان خوانا: {time.ctime()}")

# اندازه‌گیری زمان
شروع = time.time()
# کار سنگین
time.sleep(0.5)
پایان = time.time()
print(f"زمان اجرا: {پایان - شروع:.3f} ثانیه")
