با BeautifulSoup آشنا بشی، کتابخانه‌ای برای استخراج داده از صفحات وب.

📖 BeautifulSoup چیست؟

BeautifulSoup کتابخانه‌ای برای پارس کردن HTML و استخراج داده از صفحات وب هست:

  • جستجو با تگ‌ها و کلاس‌ها
  • استخراج متن و ویژگی‌ها
  • سازگاری با Requests
  • حتی صفحات نامرتب رو هم مدیریت می‌کنه

💡 نصب: pip install beautifulsoup4 lxml

💻 نمونه HTML

from bs4 import BeautifulSoup

html = """

    صفحه نمونه
    
        
سلام دنیا!

        
این یک متن نمونه است.

        

            
    
                
  • مورد ۱
    • 
                
  • مورد ۲
    • 
                
  • مورد ۳
    • 
            

            لینک
        

    

"""

soup = BeautifulSoup(html, "html.parser")

# دسترسی به تگ‌ها
print("عنوان:", soup.title)
print("متن عنوان:", soup.title.string)
print("h1:", soup.h1.text)

# پیدا کردن تگ
print("p:", soup.p.text)
print("p کلاس:", soup.p["class"])

# پیدا کردن بر اساس کلاس
p = soup.find("p", class_="content")
print("p با کلاس content:", p.text)

# پیدا کردن بر اساس id
div = soup.find(id="main")
print("div با id main:", div.text[:50])

        موجود
    



"""

soup = BeautifulSoup(html, "html.parser")

# همه محصولات
products = soup.find_all("div", class_="product")
print(f"تعداد محصولات: {len(products)}")

for product in products:
    name = product.find("h2").text
    price = product.find("span", class_="price").text
    stock = product.find("span", class_="stock").text
    print(f"{name}: {price} - {stock}")

# با CSS Selector
products = soup.select(".product")
for p in products:
    print(p.select_one("h2").text)

# با select
prices = soup.select(".price")
for price in prices:
    print(f"قیمت: {price.text}")

💻 استخراج داده از وب (مثال)

import requests
from bs4 import BeautifulSoup

# دریافت HTML از وب
url = "https://www.python.org"
response = requests.get(url)
soup = BeautifulSoup(response.text, "html.parser")

# استخراج عناوین اخبار
news_items = soup.select(".event-widget li")
print("آخرین اخبار پایتون:")

for item in news_items[:5]:
    print(f"  - {item.text.strip()}")

# استخراج لینک‌ها
links = soup.find_all("a")
print(f"\nتعداد لینک‌ها: {len(links)}")
for link in links[:5]:
    print(f"  {link.get('href', 'بدون لینک')}")

💻 پیدا کردن والدین و فرزندان

from bs4 import BeautifulSoup

html = """


    
    
        
  • مورد ۱
    • 
        
  • مورد ۲
    • 
        
  • مورد ۳
    • 
    



"""

soup = BeautifulSoup(html, "html.parser")

li = soup.find("li")
print("عنصر:", li)

# والد
print("والد:", li.parent)
print("والد بزرگ:", li.parent.parent)

# فرزندان
ul = soup.find("ul")
print("فرزندان اول:", ul.li)
print("همه فرزندان:")
for child in ul.children:
    print(f"  {child}")

# خواهر/برادر
for sibling in li.next_siblings:
    print(f"خواهر/برادر: {sibling}")

💻 اسکرپینگ پیشرفته (مثال واقعی)

import requests
from bs4 import BeautifulSoup
import time

def scrape_books():
    base_url = "http://books.toscrape.com/"
    books = []
    
    for page in range(1, 3):
        url = f"{base_url}catalogue/page-{page}.html"
        response = requests.get(url)
        soup = BeautifulSoup(response.text, "html.parser")
        
        for article in soup.select(".product_pod"):
            title = article.h3.a.get("title", "بدون عنوان")
            price = article.select_one(".price_color")
            price_text = price.text if price else "نامشخص"
            rating = article.p.get("class", ["not-rated"])[1]
            
            books.append({
                "title": title,
                "price": price_text,
                "rating": rating
            })
        
        time.sleep(1)  # احترام به سرور
    
    return books

books = scrape_books()
print(f"تعداد کتاب‌ها: {len(books)}")
for book in books[:5]:
    print(f"  {book['title']} - {book['price']} - ★{book['rating']}")
