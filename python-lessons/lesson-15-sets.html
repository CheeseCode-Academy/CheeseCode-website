🎯 درس ۱۵: مجموعه‌ها (Sets)

سطح: مبتدی | زمان مطالعه: ۱۵ دقیقه

🎯 هدف این درس

توی این درس با مجموعه‌ها آشنا میشی. مجموعه‌ها مثل کیسه‌ای هستن که اعضای تکراری ندارن و بی‌ترتیب هستن.

📖 مجموعه چیست؟

مجموعه (Set) یه ساختار داده هست که اعضای تکراری نداره و بی‌ترتیب هست. از مجموعه‌ها برای حذف اعضای تکراری و انجام عملیات‌های ریاضی روی مجموعه‌ها استفاده میشه.

فرض کن مجموعه مثل یه کیسه‌ی بین‌ترتیب از اشیاء هست که هر شیء فقط یک بار توش می‌تونه باشه.

💡 یادت باشه: مجموعه‌ها با آکولاد { } یا تابع set() تعریف میشن.

💻 تعریف مجموعه

# مجموعه خالی
مجموعه_خالی = set()  # {} برای دیکشنری خالیه!

# مجموعه از اعداد
اعداد = {1, 2, 3, 4, 5}

# مجموعه از متن‌ها
میوه‌ها = {"سیب", "موز", "پرتقال"}

# مجموعه با اعضای تکراری (تکراری‌ها حذف میشن)
تکراری = {1, 2, 2, 3, 3, 3, 4}
print(تکراری)  # {1, 2, 3, 4}

# تبدیل لیست به مجموعه (برای حذف تکراری‌ها)
لیست = [1, 2, 2, 3, 3, 4]
مجموعه = set(لیست)
print(مجموعه)  # {1, 2, 3, 4}
خروجی: 
{1, 2, 3, 4}
{1, 2, 3, 4}

💡 نکته: برای مجموعه خالی از set() استفاده کن، چون {} دیکشنری خالی می‌سازه.

💻 اضافه کردن به مجموعه

میوه‌ها = {"سیب", "موز"}

# اضافه کردن یک عضو
میوه‌ها.add("پرتقال")
print(میوه‌ها)

# اضافه کردن چند عضو
میوه‌ها.update(["انگور", "هلو", "کیوی"])
print(میوه‌ها)

# اضافه کردن عضو تکراری (تأثیری نداره)
میوه‌ها.add("سیب")
print(میوه‌ها)
خروجی: 
{'سیب', 'موز', 'پرتقال'}
{'هلو', 'سیب', 'موز', 'پرتقال', 'انگور', 'کیوی'}
{'هلو', 'سیب', 'موز', 'پرتقال', 'انگور', 'کیوی'}

💻 حذف از مجموعه

میوه‌ها = {"سیب", "موز", "پرتقال", "انگور", "هلو"}

# حذف با remove() (اگر نباشه ارور میده)
میوه‌ها.remove("موز")
print(میوه‌ها)

# حذف با discard() (اگر نباشه ارور نمیده)
میوه‌ها.discard("کیوی")  # وجود نداره، ولی ارور نمیده
print(میوه‌ها)

# حذف یک عضو دلخواه با pop()
عضو_حذف_شده = میوه‌ها.pop()
print("عضو حذف شده:", عضو_حذف_شده)
print(میوه‌ها)

# پاک کردن کل مجموعه
میوه‌ها.clear()
print(میوه‌ها)
خروجی: 
{'هلو', 'سیب', 'پرتقال', 'انگور'}
{'هلو', 'سیب', 'پرتقال', 'انگور'}
عضو حذف شده: هلو
{'سیب', 'پرتقال', 'انگور'}
set()

💻 عملیات روی مجموعه‌ها

a = {1, 2, 3, 4, 5}
b = {4, 5, 6, 7, 8}

# اشتراک (اعضای مشترک)
اشتراک = a & b
print("اشتراک:", اشتراک)

# اجتماع (همه اعضا)
اجتماع = a | b
print("اجتماع:", اجتماع)

# تفاضل (اعضای a که در b نیستن)
تفاضل = a - b
print("تفاضل a - b:", تفاضل)

# تفاضل متقارن (اعضایی که فقط در یکی هستن)
تفاضل_متقارن = a ^ b
print("تفاضل متقارن:", تفاضل_متقارن)
خروجی: 
اشتراک: {4, 5}
اجتماع: {1, 2, 3, 4, 5, 6, 7, 8}
تفاضل a - b: {1, 2, 3}
تفاضل متقارن: {1, 2, 3, 6, 7, 8}

💻 متدهای مجموعه

a = {1, 2, 3, 4, 5}
b = {4, 5}

# بررسی زیرمجموعه بودن
print(b.issubset(a))  # b زیرمجموعه‌ی a هست؟
print(a.issuperset(b))  # a بالامجموعه‌ی b هست؟

# بررسی اشتراک (آیا اشتراکی دارن؟)
print(a.isdisjoint({6, 7}))  # آیا با {6,7} اشتراکی نداره؟

# union (اجتماع)
print(a.union({6, 7}))

# intersection (اشتراک)
print(a.intersection(b))

# difference (تفاضل)
print(a.difference(b))
خروجی: 
True
True
True
{1, 2, 3, 4, 5, 6, 7}
{4, 5}
{1, 2, 3}

💻 مجموعه و حلقه for

میوه‌ها = {"سیب", "موز", "پرتقال", "انگور"}

for میوه in میوه‌ها:
    print(میوه)

# توجه: ترتیب چاپ ممکنه هر بار فرق کنه چون مجموعه بی‌ترتیبه!
خروجی (نمونه): 
موز
پرتقال
سیب
انگور

💻 مجموعه برای حذف تکراری‌ها

# حذف تکراری‌های لیست
لیست = [1, 2, 2, 3, 3, 3, 4, 4, 5]
لیست_بدون_تکراری = list(set(لیست))
print(لیست_بدون_تکراری)

# حذف کلمات تکراری در جمله
جمله = "سلام علی سلام رضا سلام"
کلمات = جمله.split()
کلمات_یکتا = set(کلمات)
print(کلمات_یکتا)
خروجی: 
[1, 2, 3, 4, 5]
{'سلام', 'علی', 'رضا'}

🧠 اشتباهات رایج

❌ اشتباه: دسترسی به اعضای مجموعه با ایندکس: مجموعه[0]

✅ راه‌حل: مجموعه‌ها بی‌ترتیب هستن و ایندکس ندارن. برای دسترسی از حلقه for استفاده کن.

❌ اشتباه: استفاده از {} برای مجموعه خالی: مجموعه = {}

✅ راه‌حل: از set() استفاده کن: مجموعه = set()

❌ اشتباه: اضافه کردن لیست به مجموعه: مجموعه.add([1, 2])

✅ راه‌حل: اعضای مجموعه باید غیرقابل تغییر باشن. از تاپل استفاده کن: مجموعه.add((1, 2))

💡 نکته‌ی هوشمندانه

از مجموعه برای پیدا کردن اعضای مشترک چند لیست استفاده کن:

لیست۱ = [1, 2, 3, 4, 5]
لیست۲ = [4, 5, 6, 7, 8]
لیست۳ = [5, 6, 7, 8, 9]

مشترک = set(لیست۱) & set(لیست۲) & set(لیست۳)
print("اعضای مشترک هر سه لیست:", مشترک)  # {5}

✍️ تمرین‌ها

تمرین ۱ (ساده): یک مجموعه از ۵ میوه بساز و با حلقه چاپ کن.

تمرین ۲ (متوسط): دو مجموعه از اعداد بساز و اشتراک، اجتماع و تفاضلشون رو پیدا کن.

تمرین ۳ (چالش‌برانگیز): برنامه‌ای بنویس که از کاربر یک جمله بگیره و کلمات تکراری رو حذف کنه و فقط کلمات یکتا رو چاپ کنه.
