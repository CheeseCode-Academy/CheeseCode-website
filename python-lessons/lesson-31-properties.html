🔒 درس ۳۱: پروپرتی‌ها (Properties)

سطح: متوسط | زمان مطالعه: ۱۸ دقیقه

🎯 هدف این درس

توی این درس با پروپرتی‌ها (Properties) آشنا میشی که بهت اجازه می‌دن به ویژگی‌های کلاس کنترل داشته باشی و کدت رو تمیزتر کنی.

📖 پروپرتی چیست؟

پروپرتی (Property) روشیه که بهت اجازه میده متدها رو مثل ویژگی (attribute) صدا بزنی. با پروپرتی می‌تونی:

  • دسترسی به ویژگی‌ها رو کنترل کنی
  • اعتبارسنجی انجام بدی
  • ویژگی‌های محاسباتی بسازی

💡 یادت باشه: پروپرتی‌ها با دکوریتور @property ساخته میشن.

💻 پروپرتی ساده

class مستطیل:
    def __init__(self, طول, عرض):
        self.طول = طول
        self.عرض = عرض
    
    @property
    def مساحت(self):
        return self.طول * self.عرض
    
    @property
    def محیط(self):
        return 2 * (self.طول + self.عرض)

م = مستطیل(5, 3)
print(f"مساحت: {م.مساحت}")  # بدون پرانتز!
print(f"محیط: {م.محیط}")    # بدون پرانتز!
خروجی: 
مساحت: 15
محیط: 16

💡 نکته: پروپرتی‌ها مثل ویژگی‌های معمولی استفاده میشن، نه مثل متد! پس بدون پرانتز صدا زده میشن.

💻 پروپرتی با setter

class شخص:
    def __init__(self, نام, سن):
        self._نام = نام
        self._سن = سن
    
    @property
    def سن(self):
        return self._سن
    
    @سن.setter
    def سن(self, مقدار):
        if مقدار < 0:
            raise ValueError("سن نمی‌تواند منفی باشد!")
        if مقدار > 120:
            raise ValueError("سن نمی‌تواند بیشتر از ۱۲۰ باشد!")
        self._سن = مقدار
    
    @property
    def نام(self):
        return self._نام
    
    @نام.setter
    def نام(self, مقدار):
        if not مقدار:
            raise ValueError("نام نمی‌تواند خالی باشد!")
        self._نام = مقدار

علی = شخص("علی", 25)
print(علی.سن)  # 25

علی.سن = 30
print(علی.سن)  # 30

# علی.سن = -5  # ValueError!

💻 پروپرتی با deleter

class حساب:
    def __init__(self):
        self._موجودی = 0
    
    @property
    def موجودی(self):
        return self._موجودی
    
    @موجودی.setter
    def موجودی(self, مقدار):
        if مقدار < 0:
            raise ValueError("موجودی نمی‌تواند منفی باشد!")
        self._موجودی = مقدار
    
    @موجودی.deleter
    def موجودی(self):
        print("حساب بسته شد!")
        del self._موجودی

حساب_من = حساب()
حساب_من.موجودی = 1000
print(حساب_من.موجودی)  # 1000

del حساب_من.موجودی  # حساب بسته شد!

💻 پروپرتی محاسباتی

class دایره:
    def __init__(self, شعاع):
        self.شعاع = شعاع
    
    @property
    def مساحت(self):
        return 3.14 * self.شعاع ** 2
    
    @property
    def محیط(self):
        return 2 * 3.14 * self.شعاع
    
    @property
    def قطر(self):
        return 2 * self.شعاع

د = دایره(5)
print(f"شعاع: {د.شعاع}")
print(f"قطر: {د.قطر}")
print(f"مساحت: {د.مساحت:.2f}")
print(f"محیط: {د.محیط:.2f}")
خروجی: 
شعاع: 5
قطر: 10
مساحت: 78.50
محیط: 31.40

💻 پروپرتی با محاسبه بر اساس ویژگی‌های دیگر

class مستطیل:
    def __init__(self, طول, عرض):
        self.طول = طول
        self.عرض = عرض
    
    @property
    def مساحت(self):
        return self.طول * self.عرض
    
    @property
    def نسبت_طول_به_عرض(self):
        return self.طول / self.عرض if self.عرض != 0 else 0
    
    @property
    def مربع_است(self):
        return self.طول == self.عرض

م = مستطیل(5, 5)
print(f"مساحت: {م.مساحت}")
print(f"نسبت: {م.نسبت_طول_به_عرض:.2f}")
print(f"مربع است: {م.مربع_است}")
خروجی: 
مساحت: 25
نسبت: 1.00
مربع است: True

💻 پروپرتی فقط خواندنی

class کتاب:
    def __init__(self, عنوان, نویسنده, سال):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
        self.سال = سال
    
    @property
    def سن(self):
        from datetime import datetime
        return datetime.now().year - self.سال
    
    @property
    def اطلاعات(self):
        return f"📚 {self.عنوان} - {self.نویسنده} ({self.سال})"

ک = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز", 1967)
print(ک.اطلاعات)
print(f"سن کتاب: {ک.سن} سال")
خروجی (نمونه): 
📚 صد سال تنهایی - گابریل گارسیا مارکز (1967)
سن کتاب: 59 سال

💻 پروپرتی با cache

class پردازش:
    def __init__(self, داده):
        self.داده = داده
        self._نتیجه_کش = None
    
    @property
    def نتیجه(self):
        if self._نتیجه_کش is None:
            print("محاسبه نتیجه...")
            self._نتیجه_کش = sum(self.داده) / len(self.داده)
        return self._نتیجه_کش

p = پردازش([10, 20, 30, 40, 50])
print(p.نتیجه)  # محاسبه نتیجه... 30.0
print(p.نتیجه)  # 30.0 (از کش استفاده شد)
خروجی: 
محاسبه نتیجه...
30.0
30.0

🧠 اشتباهات رایج

❌ اشتباه: نام‌گذاری پروپرتی با متغیر داخلی یکسان: @property def نام(self): return self.نام

✅ راه‌حل: از _نام برای متغیر داخلی استفاده کن.

❌ اشتباه: فراموش کردن @property برای getter

✅ راه‌حل: حتماً قبل از getter از @property استفاده کن.

❌ اشتباه: استفاده از @property برای متدهایی که پارامتر دارن

✅ راه‌حل: پروپرتی‌ها فقط برای متدهای بدون پارامتر (به جز self) هستن.

💡 نکته‌ی هوشمندانه

از @property برای سازگاری با نسخه‌های قبلی استفاده کن. می‌تونی ویژگی رو به پروپرتی تبدیل کنی بدون اینکه کدهای دیگه رو تغییر بدی:

class شخص:
    def __init__(self, نام):
        self._نام = نام
    
    @property
    def نام(self):
        return self._نام
    
    @نام.setter
    def نام(self, مقدار):
        self._نام = مقدار

# کد قدیمی که مستقیم به نام دسترسی داشت، همچنان کار می‌کنه!

✍️ تمرین‌ها
