📝 درس ۳۶: تایپ‌هینت‌ها (Type Hints)

سطح: متوسط | زمان مطالعه: ۱۸ دقیقه

🎯 هدف این درس

توی این درس با تایپ‌هینت‌ها آشنا میشی که کدت رو خوانا‌تر و قابل فهم‌تر می‌کنن و به IDEها کمک می‌کنن خطاها رو زودتر پیدا کنن.

📖 تایپ‌هینت چیست؟

تایپ‌هینت (Type Hint) مشخص می‌کنه که یه متغیر یا تابع چه نوعی از داده رو دریافت می‌کنه یا برمی‌گردونه.

تایپ‌هینت‌ها اجباری نیستن و فقط برای مستندسازی و کمک به IDE هستن.

💡 یادت باشه: تایپ‌هینت‌ها در پایتون ۳.۵+ اضافه شدن.

💻 تایپ‌هینت ساده

def جمع(a: int, b: int) -> int:
    return a + b

def سلام(نام: str) -> str:
    return f"سلام {نام}!"

# استفاده
نتیجه: int = جمع(5, 3)
پیام: str = سلام("علی")

print(نتیجه)  # 8
print(پیام)   # سلام علی!

💻 تایپ‌های مختلف

from typing import List, Dict, Tuple, Optional, Union

# لیست
def پردازش_اعداد(اعداد: List[int]) -> List[int]:
    return [x * 2 for x in اعداد]

# دیکشنری
def اطلاعات_فرد(فرد: Dict[str, Union[str, int]]) -> str:
    return f"{فرد['نام']} - {فرد['سن']} سال"

# تاپل
def مختصات() -> Tuple[int, int]:
    return (10, 20)

# Optional (می‌تونه None باشه)
def نام(شخص: Optional[str] = None) -> str:
    if شخص:
        return f"سلام {شخص}!"
    return "سلام مهمان!"

# Union (چند نوع)
def تبدیل(مقدار: Union[int, float, str]) -> str:
    return str(مقدار)

💻 تایپ‌های پیشرفته

from typing import Callable, Any, TypeVar, Generic

# Callable (تابع)
def اجرا_کننده(تابع: Callable[[int, int], int], a: int, b: int) -> int:
    return تابع(a, b)

# Any (هر نوعی)
def پردازش_هر(داده: Any) -> None:
    print(داده)

# TypeVar (نوع متغیر)
T = TypeVar('T')

def اولین(لیست: List[T]) -> T:
    return لیست[0]

# Generic
from typing import Generic

class Stack(Generic[T]):
    def __init__(self) -> None:
        self._items: List[T] = []
    
    def push(self, item: T) -> None:
        self._items.append(item)
    
    def pop(self) -> T:
        return self._items.pop()

💻 تایپ‌هینت برای کلاس‌ها

class شخص:
    def __init__(self, نام: str, سن: int) -> None:
        self.نام = نام
        self.سن = سن
    
    def معرفی(self) -> str:
        return f"من {self.نام} هستم"

class دانشجو(شخص):
    def __init__(self, نام: str, سن: int, رشته: str) -> None:
        super().__init__(نام, سن)
        self.رشته = رشته
    
    def معرفی(self) -> str:
        return f"{super().معرفی()} و رشته‌ام {self.رشته} است"

💻 نوع‌های سفارشی

from typing import NewType

# ایجاد نوع سفارشی
UserId = NewType('UserId', int)
ProductId = NewType('ProductId', str)

def get_user(user_id: UserId) -> str:
    return f"کاربر {user_id}"

def get_product(product_id: ProductId) -> str:
    return f"محصول {product_id}"

# استفاده
user_id = UserId(123)
product_id = ProductId("P-456")

print(get_user(user_id))
print(get_product(product_id))

💻 Literal و Final

from typing import Literal, Final

# Literal - مقادیر مشخص
def وضعیت(وضع: Literal["فعال", "غیرفعال"]) -> str:
    return f"وضعیت: {وضع}"

# Final - ثابت
MAX_SIZE: Final = 100
PI: Final[float] = 3.14159

print(وضعیت("فعال"))    # وضعیت: فعال

💡 نکته‌ی هوشمندانه

از TypeAlias برای ساده‌سازی تایپ‌های پیچیده استفاده کن:

from typing import TypeAlias, Dict, List, Union

PersonDict: TypeAlias = Dict[str, Union[str, int]]
PeopleList: TypeAlias = List[PersonDict]

def پردازش_افراد(افراد: PeopleList) -> None:
    for شخص in افراد:
        print(شخص)

# استفاده
افراد = [{"نام": "علی", "سن": 25}, {"نام": "زهرا", "سن": 30}]
پردازش_افراد(افراد)

✍️ تمرین‌ها

تمرین ۱ (ساده): تابعی با تایپ‌هینت بنویس که دو عدد رو جمع کنه و نتیجه رو برگردونه.

تمرین ۲ (متوسط): تابعی با تایپ‌هینت بنویس که لیستی از اعداد بگیره و مجموعشون رو برگردونه.

تمرین ۳ (چالش‌برانگیز): یک کلاس با تایپ‌هینت کامل بنویس که اطلاعات یک کتاب رو مدیریت کنه.
