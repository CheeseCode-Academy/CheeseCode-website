سطح: متوسط | زمان مطالعه: ۱۸ دقیقه

🎯 هدف این درس

توی این درس با متدهای خاص یا جادویی (Magic Methods) آشنا میشی که به کلاس‌ها قابلیت‌های ویژه‌ای می‌دن.

📖 متدهای خاص چیست؟

متدهای خاص (Magic Methods) متدهایی هستن که با دو زیرخط شروع و تموم میشن (مثل __init__). این متدها به کلاس‌ها اجازه می‌دن رفتارهای ویژه‌ای داشته باشن.

💡 یادت باشه: متدهای خاص به‌طور خودکار در شرایط خاص فراخوانی میشن.

💻 __init__ (سازنده)

class شخص:
    def __init__(self, نام):
        self.نام = نام
        print(f"{self.نام} ساخته شد!")

علی = شخص("علی")  # علی ساخته شد!

💻 __str__ و __repr__

class کتاب:
    def __init__(self, عنوان, نویسنده):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
    
    def __str__(self):
        return f"📚 {self.عنوان} - {self.نویسنده}"
    
    def __repr__(self):
        return f"کتاب('{self.عنوان}', '{self.نویسنده}')"

ک = کتاب("بوف کور", "صادق هدایت")
print(ک)  # 📚 بوف کور - صادق هدایت
print(repr(ک))  # کتاب('بوف کور', 'صادق هدایت')

💻 __len__ و __getitem__

class لیست_سفارشی:
    def __init__(self, *items):
        self.items = list(items)
    
    def __len__(self):
        return len(self.items)
    
    def __getitem__(self, index):
        return self.items[index]
    
    def __setitem__(self, index, value):
        self.items[index] = value

لیست = لیست_سفارشی(1, 2, 3, 4, 5)
print(len(لیست))  # 5
print(لیست[2])    # 3
لیست[2] = 10
print(لیست[2])    # 10

💻 __iter__ و __next__

class شمارنده:
    def __init__(self, شروع, پایان):
        self.شروع = شروع
        self.پایان = پایان
    
    def __iter__(self):
        self.فعلی = self.شروع
        return self
    
    def __next__(self):
        if self.فعلی > self.پایان:
            raise StopIteration
        مقدار = self.فعلی
        self.فعلی += 1
        return مقدار

for i in شمارنده(1, 5):
    print(i, end=" ")  # 1 2 3 4 5

💻 __call__

شیء رو قابل صدا زدن می‌کنه:

class ضرب_کننده:
    def __init__(self, ضریب):
        self.ضریب = ضریب
    
    def __call__(self, عدد):
        return عدد * self.ضریب

ضرب_در_دو = ضرب_کننده(2)
print(ضرب_در_دو(5))   # 10
print(ضرب_در_دو(10))  # 20

💻 عملگرهای ریاضی

class عدد_خاص:
    def __init__(self, مقدار):
        self.مقدار = مقدار
    
    def __add__(self, دیگر):
        return عدد_خاص(self.مقدار + دیگر.مقدار)
    
    def __sub__(self, دیگر):
        return عدد_خاص(self.مقدار - دیگر.مقدار)
    
    def __mul__(self, دیگر):
        return عدد_خاص(self.مقدار * دیگر.مقدار)
    
    def __str__(self):
        return str(self.مقدار)

a = عدد_خاص(10)
b = عدد_خاص(5)
print(a + b)  # 15
print(a - b)  # 5
print(a * b)  # 50

💻 عملگرهای مقایسه‌ای

class فرد:
    def __init__(self, نام, سن):
        self.نام = نام
        self.سن = سن
    
    def __eq__(self, دیگر):
        return self.سن == دیگر.سن
    
    def __lt__(self, دیگر):
        return self.سن < دیگر.سن
    
    def __gt__(self, دیگر):
        return self.سن > دیگر.سن
    
    def __str__(self):
        return f"{self.نام} ({self.سن})"

علی = فرد("علی", 25)
زهرا = فرد("زهرا", 30)
رضا = فرد("رضا", 25)

print(علی == زهرا)  # False
print(علی == رضا)   # True
print(علی < زهرا)   # True
print(علی > زهرا)   # False

💻 __enter__ و __exit__

برای استفاده با with:

class مدیریت_فایل:
    def __init__(self, نام_فایل):
        self.نام_فایل = نام_فایل
    
    def __enter__(self):
        self.فایل = open(self.نام_فایل, "w")
        return self.فایل
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.فایل.close()
        print("فایل بسته شد")

with مدیریت_فایل("test.txt") as فایل:
    فایل.write("سلام!")

# فایل بسته شد

💻 __dir__ و __dict__

class مثال:
    def __init__(self):
        self.a = 1
        self.b = 2
    
    def __dir__(self):
        return ["a", "b"]

م = مثال()
print(dir(م))  # ['a', 'b']
print(م.__dict__)  # {'a': 1, 'b': 2}

🧠 اشتباهات رایج

❌ اشتباه: فراموش کردن return در متدهای خاص:

def __add__(self, دیگر):
    self.مقدار + دیگر.مقدار  # return رو فراموش کردی!

✅ راه‌حل: حتماً return کن.

❌ اشتباه: استفاده از متدهای خاص برای کارهای غیرمرتبط

✅ راه‌حل: هر متد خاص یه کار مشخص داره، ازش برای همون کار استفاده کن.

💡 نکته‌ی هوشمندانه

از __getattr__ و __setattr__ برای مدیریت پویا ویژگی‌ها استفاده کن:

class پویا:
    def __getattr__(self, نام):
        return f"ویژگی {نام} وجود ندارد!"
    
    def __setattr__(self, نام, مقدار):
        if نام == "سن" and مقدار < 0:
            raise ValueError("سن نمی‌تواند منفی باشد!")
        super().__setattr__(نام, مقدار)

obj = پویا()
print(obj.چیزی)  # ویژگی چیزی وجود ندارد!
obj.سن = 25
# obj.سن = -5  # ValueError!

✍️ تمرین‌ها

تمرین ۳ (چالش‌برانگیز): کلاسی برای زمان بساز که با __sub__ اختلاف دو زمان رو حساب کنه.
