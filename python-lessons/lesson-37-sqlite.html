🗄️ درس ۳۷: کار با SQLite

سطح: پیشرفته | زمان مطالعه: ۲۵ دقیقه

🎯 هدف این درس

توی این درس با SQLite آشنا میشی که یک دیتابیس سبک و بدون نیاز به سرور هست. یاد می‌گیری چطور داده‌ها رو توی دیتابیس ذخیره و مدیریت کنی.

📖 SQLite چیست؟

SQLite یک دیتابیس ساده و سبک هست که به‌صورت فایل ذخیره میشه و نیازی به نصب سرور نداره.

  • بدون سرور
  • فایل واحد
  • سبک و سریع
  • در پایتون داخلی‌ست

💡 یادت باشه: SQLite با ماژول sqlite3 در پایتون کار می‌کنه.

💻 اتصال به دیتابیس

import sqlite3

# اتصال به دیتابیس (ایجاد میشه اگر نباشه)
conn = sqlite3.connect("mydb.db")

# ایجاد یک cursor برای اجرای دستورات
cursor = conn.cursor()

# بستن اتصال
conn.close()

💻 ایجاد جدول

import sqlite3

conn = sqlite3.connect("mydb.db")
cursor = conn.cursor()

# ایجاد جدول کاربران
cursor.execute("""
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        age INTEGER,
        city TEXT
    )
""")

conn.commit()
conn.close()
print("جدول ساخته شد!")

💻 درج داده

import sqlite3

conn = sqlite3.connect("mydb.db")
cursor = conn.cursor()

# درج یک رکورد
cursor.execute("""
    INSERT INTO users (name, age, city)
    VALUES (?, ?, ?)
""", ("علی", 25, "تهران"))

# درج چند رکورد
users = [
    ("زهرا", 30, "اصفهان"),
    ("رضا", 22, "شیراز"),
    ("سارا", 28, "مشهد")
]

cursor.executemany("""
    INSERT INTO users (name, age, city)
    VALUES (?, ?, ?)
""", users)

conn.commit()
conn.close()
print("داده‌ها ذخیره شدند!")

💻 خواندن داده

import sqlite3

conn = sqlite3.connect("mydb.db")
cursor = conn.cursor()

# خواندن همه
cursor.execute("SELECT * FROM users")
همه = cursor.fetchall()
print("همه کاربران:")
for user in همه:
    print(f"  ID: {user[0]}, نام: {user[1]}, سن: {user[2]}, شهر: {user[3]}")

# خواندن با شرط
cursor.execute("SELECT * FROM users WHERE age > ?", (25,))
بزرگتر = cursor.fetchall()
print(f"\nکاربران بالای ۲۵ سال: {len(بزرگتر)} نفر")

# خواندن یک رکورد
cursor.execute("SELECT * FROM users WHERE name = ?", ("علی",))
علی = cursor.fetchone()
if علی:
    print(f"\nعلی: {علی}")

conn.close()

💻 به‌روزرسانی داده

import sqlite3

conn = sqlite3.connect("mydb.db")
cursor = conn.cursor()

# به‌روزرسانی
cursor.execute("""
    UPDATE users
    SET city = ?
    WHERE name = ?
""", ("قم", "علی"))

conn.commit()
print("به‌روزرسانی انجام شد!")

# بررسی
cursor.execute("SELECT * FROM users WHERE name = 'علی'")
print(cursor.fetchone())

conn.close()

💻 کار با کلاس

import sqlite3
from typing import List, Optional, Dict

class Database:
    def __init__(self, db_name: str = "mydb.db"):
        self.db_name = db_name
        self._init_table()
    
    def _init_table(self):
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("""
                CREATE TABLE IF NOT EXISTS users (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    name TEXT NOT NULL,
                    age INTEGER,
                    city TEXT
                )
            """)
            conn.commit()
    
    def add_user(self, name: str, age: int, city: str) -> int:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("""
                INSERT INTO users (name, age, city)
                VALUES (?, ?, ?)
            """, (name, age, city))
            conn.commit()
            return cursor.lastrowid
    
    def get_all_users(self) -> List[Dict]:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM users")
            rows = cursor.fetchall()
            return [{"id": r[0], "name": r[1], "age": r[2], "city": r[3]} for r in rows]
    
    def update_user(self, user_id: int, **kwargs) -> bool:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            for key, value in kwargs.items():
                cursor.execute(f"UPDATE users SET {key} = ? WHERE id = ?", (value, user_id))
            conn.commit()
            return True
    
    def delete_user(self, user_id: int) -> bool:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("DELETE FROM users WHERE id = ?", (user_id,))
            conn.commit()
            return True

# استفاده
db = Database()
db.add_user("علی", 25, "تهران")
db.add_user("زهرا", 30, "اصفهان")

users = db.get_all_users()
for user in users:
    print(user)

db.update_user(1, city="قم")
db.delete_user(2)

💻 کار با SQLite و pandas

import sqlite3
import pandas as pd

# خواندن دیتابیس به pandas
conn = sqlite3.connect("mydb.db")
df = pd.read_sql_query("SELECT * FROM users", conn)
print(df)

# ذخیره pandas به دیتابیس
new_data = pd.DataFrame([
    {"name": "رضا", "age": 22, "city": "شیراز"},
    {"name": "سارا", "age": 28, "city": "مشهد"}
])
new_data.to_sql("users", conn, if_exists="append", index=False)
conn.close()

💡 نکته‌ی هوشمندانه

از with برای مدیریت خودکار اتصال استفاده کن:

import sqlite3

with sqlite3.connect("mydb.db") as conn:
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM users")
    # اتصال به‌طور خودکار بسته میشه

برای خطاگیری از try/except استفاده کن:

try:
    with sqlite3.connect("mydb.db") as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT * FROM users")
except sqlite3.Error as e:
    print(f"خطای دیتابیس: {e}")

✍️ تمرین‌ها

تمرین ۱ (ساده): یک دیتابیس با جدول products (نام، قیمت، تعداد) بساز.

تمرین ۲ (متوسط): به جدول products چند محصول اضافه کن و همه رو بخون.

تمرین ۳ (چالش‌برانگیز): یک کلاس کامل برای مدیریت محصولات بساز (اضافه، حذف، به‌روزرسانی، جستجو).
