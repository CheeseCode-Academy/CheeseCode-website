📝 پاسخ تمرین ۱

سوال: با datetime تاریخ تولد خودت رو وارد کن و سنت رو حساب کن.

✅ پاسخ:

import datetime

تاریخ_تولد = datetime.datetime(2000, 1, 15)  # نمونه
اکنون = datetime.datetime.now()

سن = اکنون - تاریخ_تولد
سال‌ها = سن.days // 365
روزها = سن.days % 365

print(f"سن شما: {سال‌ها} سال و {روزها} روز")
خروجی (نمونه): 
سن شما: 26 سال و 187 روز

📝 پاسخ تمرین ۳

سوال: با os یک اسکریپت بنویس که همه‌ی فایل‌های .txt توی یک پوشه رو پیدا کنه و اندازه‌شون رو چاپ کنه.

✅ پاسخ:

import os

def پیدا_کن_فایل‌های_txt(پوشه):
    if not os.path.exists(پوشه):
        print(f"پوشه '{پوشه}' وجود ندارد!")
        return
    
    if not os.path.isdir(پوشه):
        print(f"'{پوشه}' یک پوشه نیست!")
        return
    
    فایل‌ها = os.listdir(پوشه)
    txt_فایل‌ها = [f for f in فایل‌ها if f.endswith(".txt")]
    
    if not txt_فایل‌ها:
        print(f"هیچ فایل .txt در پوشه '{پوشه}' یافت نشد!")
        return
    
    print(f"فایل‌های .txt در '{پوشه}':")
    for فایل in txt_فایل‌ها:
        مسیر = os.path.join(پوشه, فایل)
        اندازه = os.path.getsize(مسیر)
        print(f"  - {فایل}: {اندازه:,} بایت")

# تست با پوشه فعلی
پیدا_کن_فایل‌های_txt(".")
خروجی (نمونه): 
فایل‌های .txt در '.':
  - file1.txt: 1,024 بایت
  - file2.txt: 2,048 بایت
