📝 پاسخ تمرین ۱

سوال: کلاسی به اسم مستطیل بساز که طول و عرض رو بگیره و مساحت و محیط رو حساب کنه.

✅ پاسخ:

class مستطیل:
    def __init__(self, طول, عرض):
        self.طول = طول
        self.عرض = عرض
    
    def مساحت(self):
        return self.طول * self.عرض
    
    def محیط(self):
        return 2 * (self.طول + self.عرض)

م = مستطیل(5, 3)
print(f"مساحت: {م.مساحت()}")
print(f"محیط: {م.محیط()}")
خروجی: 
مساحت: 15
محیط: 16

📝 پاسخ تمرین ۲

سوال: کلاسی به اسم دانشجو بساز با ویژگی‌های نام، شماره دانشجویی و نمرات. متدی برای محاسبه معدل اضافه کن.

✅ پاسخ:

class دانشجو:
    def __init__(self, نام, شماره, نمرات=None):
        self.نام = نام
        self.شماره = شماره
        self.نمرات = نمرات or []
    
    def اضافه_نمره(self, نمره):
        self.نمرات.append(نمره)
    
    def معدل(self):
        if not self.نمرات:
            return 0
        return sum(self.نمرات) / len(self.نمرات)
    
    def نمایش(self):
        return f"{self.نام} (شماره: {self.شماره}) - معدل: {self.معدل():.2f}"

علی = دانشجو("علی", "12345")
علی.اضافه_نمره(18)
علی.اضافه_نمره(16)
علی.اضافه_نمره(19)

print(علی.نمایش())
خروجی: 
علی (شماره: 12345) - معدل: 17.67

📝 پاسخ تمرین ۳

سوال: کلاسی به اسم بانک بساز که حساب‌های کاربری رو مدیریت کنه.

✅ پاسخ:

class بانک:
    def __init__(self):
        self.حساب‌ها = {}
    
    def ایجاد_حساب(self, شماره, موجودی=0):
        if شماره in self.حساب‌ها:
            return "حساب قبلاً وجود دارد!"
        self.حساب‌ها[شماره] = موجودی
        return "حساب ایجاد شد!"
    
    def واریز(self, شماره, مبلغ):
        if شماره not in self.حساب‌ها:
            return "حساب پیدا نشد!"
        if مبلغ <= 0:
            return "مبلغ باید مثبت باشد!"
        self.حساب‌ها[شماره] += مبلغ
        return f"{مبلغ} تومان واریز شد. موجودی: {self.حساب‌ها[شماره]}"
    
    def برداشت(self, شماره, مبلغ):
        if شماره not in self.حساب‌ها:
            return "حساب پیدا نشد!"
        if مبلغ <= 0:
            return "مبلغ باید مثبت باشد!"
        if self.حساب‌ها[شماره] < مبلغ:
            return "موجودی کافی نیست!"
        self.حساب‌ها[شماره] -= مبلغ
        return f"{مبلغ} تومان برداشت شد. موجودی: {self.حساب‌ها[شماره]}"
    
    def موجودی(self, شماره):
        if شماره not in self.حساب‌ها:
            return "حساب پیدا نشد!"
        return f"موجودی: {self.حساب‌ها[شماره]}"

بانک_ملی = بانک()
print(بانک_ملی.ایجاد_حساب("123", 1000))
print(بانک_ملی.واریز("123", 500))
print(بانک_ملی.برداشت("123", 200))
print(بانک_ملی.موجودی("123"))
خروجی: 
حساب ایجاد شد!
500 تومان واریز شد. موجودی: 1500
200 تومان برداشت شد. موجودی: 1300
موجودی: 1300
