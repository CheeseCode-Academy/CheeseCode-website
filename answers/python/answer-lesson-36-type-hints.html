📝 پاسخ تمرین ۲

سوال: تابعی با تایپ‌هینت بنویس که لیستی از اعداد بگیره و مجموعشون رو برگردونه.

✅ پاسخ:

from typing import List

def مجموع_لیست(اعداد: List[int]) -> int:
    return sum(اعداد)

print(مجموع_لیست([10, 20, 30]))  # 60

📝 پاسخ تمرین ۳

سوال: یک کلاس با تایپ‌هینت کامل بنویس که اطلاعات یک کتاب رو مدیریت کنه.

✅ پاسخ:

from typing import Optional, List

class کتاب:
    def __init__(self, عنوان: str, نویسنده: str, سال: int, شابک: str) -> None:
        self.عنوان: str = عنوان
        self.نویسنده: str = نویسنده
        self.سال: int = سال
        self.شابک: str = شابک
        self.امانت_رفته: bool = False
        self.امانت_گیرنده: Optional[str] = None
    
    def امانت_بده(self, گیرنده: str) -> str:
        if self.امانت_رفته:
            return f"کتاب '{self.عنوان}' قبلاً امانت رفته است!"
        self.امانت_رفته = True
        self.امانت_گیرنده = گیرنده
        return f"کتاب '{self.عنوان}' به {گیرنده} امانت داده شد"
    
    def بازگردان(self) -> str:
        if not self.امانت_رفته:
            return f"کتاب '{self.عنوان}' در کتابخانه است!"
        self.امانت_رفته = False
        self.امانت_گیرنده = None
        return f"کتاب '{self.عنوان}' بازگردانده شد"
    
    def __str__(self) -> str:
        وضعیت = "📕 موجود" if not self.امانت_رفته else f"📗 امانت: {self.امانت_گیرنده}"
        return f"{self.عنوان} - {self.نویسنده} ({self.سال}) [{وضعیت}]"

ک = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز", 1967, "978-123")
print(ک)
print(ک.امانت_بده("علی"))
print(ک)
