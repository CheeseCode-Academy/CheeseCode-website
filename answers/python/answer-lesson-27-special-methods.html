📝 پاسخ تمرین ۱

سوال: کلاسی بساز که با __str__ اطلاعات رو نمایش بده.

✅ پاسخ:

class محصول:
    def __init__(self, نام, قیمت):
        self.نام = نام
        self.قیمت = قیمت
    
    def __str__(self):
        return f"📦 {self.نام} - {self.قیمت:,} تومان"

کالا = محصول("لپ‌تاپ", 15000000)
print(کالا)
خروجی: 
📦 لپ‌تاپ - 15,000,000 تومان

📝 پاسخ تمرین ۲

سوال: کلاسی برای برداری بساز که جمع و تفریق رو با عملگرها پشتیبانی کنه.

✅ پاسخ:

class بردار:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, دیگر):
        return بردار(self.x + دیگر.x, self.y + دیگر.y)
    
    def __sub__(self, دیگر):
        return بردار(self.x - دیگر.x, self.y - دیگر.y)
    
    def __str__(self):
        return f"({self.x}, {self.y})"

v1 = بردار(3, 4)
v2 = بردار(1, 2)
print(v1 + v2)  # (4, 6)
print(v1 - v2)  # (2, 2)
خروجی: 
(4, 6)
(2, 2)

📝 پاسخ تمرین ۳

سوال: کلاسی برای زمان بساز که با __sub__ اختلاف دو زمان رو حساب کنه.

✅ پاسخ:

class زمان:
    def __init__(self, ساعت, دقیقه=0):
        self.ساعت = ساعت
        self.دقیقه = دقیقه
    
    def __sub__(self, دیگر):
        دقیقه‌ها_کل = (self.ساعت * 60 + self.دقیقه) - (دیگر.ساعت * 60 + دیگر.دقیقه)
        ساعت = دقیقه‌ها_کل // 60
        دقیقه = دقیقه‌ها_کل % 60
        return زمان(ساعت, دقیقه)
    
    def __str__(self):
        return f"{self.ساعت}:{self.دقیقه:02d}"

t1 = زمان(10, 30)
t2 = زمان(8, 45)
اختلاف = t1 - t2
print(f"اختلاف: {اختلاف}")  # 1:45
خروجی: 
اختلاف: 1:45
