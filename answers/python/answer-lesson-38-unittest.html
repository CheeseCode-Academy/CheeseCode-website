📝 پاسخ تمرین ۱

سوال: برای تابع ضرب یک تست بنویس.

✅ پاسخ:

import unittest

def ضرب(a, b):
    return a * b

class Testضرب(unittest.TestCase):
    def test_ضرب_اعداد_مثبت(self):
        self.assertEqual(ضرب(5, 3), 15)
        self.assertEqual(ضرب(10, 2), 20)
    
    def test_ضرب_با_صفر(self):
        self.assertEqual(ضرب(5, 0), 0)
        self.assertEqual(ضرب(0, 10), 0)
    
    def test_ضرب_اعداد_منفی(self):
        self.assertEqual(ضرب(-5, 3), -15)
        self.assertEqual(ضرب(-5, -3), 15)

if __name__ == "__main__":
    unittest.main()

📝 پاسخ تمرین ۲

سوال: برای کلاس مستطیل (مساحت و محیط) تست بنویس.

✅ پاسخ:

import unittest

class مستطیل:
    def __init__(self, طول, عرض):
        self.طول = طول
        self.عرض = عرض
    
    def مساحت(self):
        return self.طول * self.عرض
    
    def محیط(self):
        return 2 * (self.طول + self.عرض)

class Testمستطیل(unittest.TestCase):
    def setUp(self):
        self.م = مستطیل(5, 3)
        self.مربع = مستطیل(4, 4)
    
    def test_مساحت(self):
        self.assertEqual(self.م.مساحت(), 15)
        self.assertEqual(self.مربع.مساحت(), 16)
    
    def test_محیط(self):
        self.assertEqual(self.م.محیط(), 16)
        self.assertEqual(self.مربع.محیط(), 16)
    
    def test_اعداد_صفر(self):
        صفر = مستطیل(0, 5)
        self.assertEqual(صفر.مساحت(), 0)
        self.assertEqual(صفر.محیط(), 10)

if __name__ == "__main__":
    unittest.main()

📝 پاسخ تمرین ۳

سوال: برای کلاس کتابخانه (که در درس ۳۲ ساختیم) تست بنویس.

✅ پاسخ:

import unittest

class کتاب:
    def __init__(self, عنوان, نویسنده, سال, شابک):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
        self.سال = سال
        self.شابک = شابک
        self.امانت_رفته = False
        self.امانت_گیرنده = None
    
    def امانت_بده(self, عضو):
        if self.امانت_رفته:
            return "کتاب قبلاً امانت رفته است!"
        self.امانت_رفته = True
        self.امانت_گیرنده = عضو
        return f"کتاب '{self.عنوان}' به {عضو} امانت داده شد"
    
    def بازگردان(self):
        if not self.امانت_رفته:
            return "کتاب در کتابخانه است!"
        self.امانت_رفته = False
        self.امانت_گیرنده = None
        return f"کتاب '{self.عنوان}' بازگردانده شد"

class کتابخانه:
    def __init__(self):
        self.کتاب‌ها = []
    
    def اضافه_کتاب(self, کتاب):
        self.کتاب‌ها.append(کتاب)
        return f"کتاب '{کتاب.عنوان}' اضافه شد"
    
    def پیدا_کن_کتاب(self, شابک):
        for کتاب in self.کتاب‌ها:
            if کتاب.شابک == شابک:
                return کتاب
        return None

class Testکتابخانه(unittest.TestCase):
    def setUp(self):
        self.کتابخانه = کتابخانه()
        self.کتاب1 = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز", 1967, "978-123")
        self.کتاب2 = کتاب("بوف کور", "صادق هدایت", 1937, "978-456")
        self.کتابخانه.اضافه_کتاب(self.کتاب1)
        self.کتابخانه.اضافه_کتاب(self.کتاب2)
    
    def test_اضافه_کتاب(self):
        self.assertEqual(len(self.کتابخانه.کتاب‌ها), 2)
    
    def test_پیدا_کن_کتاب(self):
        کتاب = self.کتابخانه.پیدا_کن_کتاب("978-123")
        self.assertIsNotNone(کتاب)
        self.assertEqual(کتاب.عنوان, "صد سال تنهایی")
    
    def test_امانت_کتاب(self):
        result = self.کتاب1.امانت_بده("علی")
        self.assertEqual(result, "کتاب 'صد سال تنهایی' به علی امانت داده شد")
        self.assertTrue(self.کتاب1.امانت_رفته)
    
    def test_بازگرداندن_کتاب(self):
        self.کتاب1.امانت_بده("علی")
        result = self.کتاب1.بازگردان()
        self.assertEqual(result, "کتاب 'صد سال تنهایی' بازگردانده شد")
        self.assertFalse(self.کتاب1.امانت_رفته)

if __name__ == "__main__":
    unittest.main()
