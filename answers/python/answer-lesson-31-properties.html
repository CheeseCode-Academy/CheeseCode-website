📝 پاسخ تمرین ۱

سوال: کلاسی با پروپرتی مربع بساز که مقدار رو به توان ۲ برسونه.

✅ پاسخ:

class عدد:
    def __init__(self, مقدار):
        self.مقدار = مقدار
    
    @property
    def مربع(self):
        return self.مقدار ** 2

n = عدد(5)
print(f"مربع ۵: {n.مربع}")
خروجی: 
مربع ۵: 25

📝 پاسخ تمرین ۲

سوال: کلاس دانشجو با پروپرتی معدل بساز که از لیست نمرات محاسبه بشه.

✅ پاسخ:

class دانشجو:
    def __init__(self, نام, نمرات=None):
        self.نام = نام
        self._نمرات = نمرات or []
    
    def اضافه_نمره(self, نمره):
        self._نمرات.append(نمره)
    
    @property
    def معدل(self):
        if not self._نمرات:
            return 0
        return sum(self._نمرات) / len(self._نمرات)
    
    @property
    def تعداد_درس(self):
        return len(self._نمرات)

علی = دانشجو("علی")
علی.اضافه_نمره(18)
علی.اضافه_نمره(16)
علی.اضافه_نمره(19)

print(f"معدل علی: {علی.معدل:.2f}")
print(f"تعداد درس: {علی.تعداد_درس}")
خروجی: 
معدل علی: 17.67
تعداد درس: 3

📝 پاسخ تمرین ۳

سوال: کلاس حساب_بانکی با پروپرتی موجودی بساز که برداشت و واریز رو کنترل کنه.

✅ پاسخ:

class حساب_بانکی:
    def __init__(self, نام, موجودی_اولیه=0):
        self.نام = نام
        self._موجودی = موجودی_اولیه
    
    @property
    def موجودی(self):
        return self._موجودی
    
    @موجودی.setter
    def موجودی(self, مقدار):
        if مقدار < 0:
            raise ValueError("موجودی نمی‌تواند منفی باشد!")
        self._موجودی = مقدار
    
    def واریز(self, مبلغ):
        if مبلغ <= 0:
            raise ValueError("مبلغ واریز باید مثبت باشد!")
        self._موجودی += مبلغ
        return f"{مبلغ} تومان واریز شد. موجودی: {self._موجودی}"
    
    def برداشت(self, مبلغ):
        if مبلغ <= 0:
            raise ValueError("مبلغ برداشت باید مثبت باشد!")
        if مبلغ > self._موجودی:
            raise ValueError("موجودی کافی نیست!")
        self._موجودی -= مبلغ
        return f"{مبلغ} تومان برداشت شد. موجودی: {self._موجودی}"

حساب = حساب_بانکی("علی", 1000)
print(حساب.واریز(500))
print(حساب.برداشت(200))
print(f"موجودی نهایی: {حساب.موجودی}")
خروجی: 
500 تومان واریز شد. موجودی: 1500
200 تومان برداشت شد. موجودی: 1300
موجودی نهایی: 1300
