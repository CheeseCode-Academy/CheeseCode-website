📝 پاسخ تمرین ۱

سوال: یک DataFrame با اطلاعات ۵ نفر بساز (نام، سن، شهر، حقوق) و آمار پایه رو نمایش بده.

✅ پاسخ:

import pandas as pd

data = {
    "نام": ["علی", "زهرا", "رضا", "سارا", "مهدی"],
    "سن": [25, 30, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", "تهران", "مشهد", "شیراز"],
    "حقوق": [5000, 6000, 4500, 7000, 8000]
}
df = pd.DataFrame(data)

print("DataFrame:\n", df)
print("\nآمار:\n", df.describe())
print("\nاطلاعات:\n", df.info())

📝 پاسخ تمرین ۲

سوال: DataFrame رو بر اساس سن مرتب کن و فقط افرادی که حقوق > ۵۰۰۰ هستن رو نمایش بده.

✅ پاسخ:

import pandas as pd

data = {
    "نام": ["علی", "زهرا", "رضا", "سارا", "مهدی"],
    "سن": [25, 30, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", "تهران", "مشهد", "شیراز"],
    "حقوق": [5000, 6000, 4500, 7000, 8000]
}
df = pd.DataFrame(data)

# مرتب‌سازی
sorted_df = df.sort_values("سن")
print("مرتب‌سازی بر اساس سن:\n", sorted_df)

# فیلتر
high_salary = df[df["حقوق"] > 5000]
print("\nافراد با حقوق > ۵۰۰۰:\n", high_salary)

📝 پاسخ تمرین ۳

سوال: DataFrame رو بر اساس شهر گروه‌بندی کن و میانگین حقوق هر شهر رو حساب کن.

✅ پاسخ:

import pandas as pd

data = {
    "نام": ["علی", "زهرا", "رضا", "سارا", "مهدی"],
    "سن": [25, 30, 22, 28, 35],
    "شهر": ["تهران", "اصفهان", "تهران", "مشهد", "شیراز"],
    "حقوق": [5000, 6000, 4500, 7000, 8000]
}
df = pd.DataFrame(data)

# گروه‌بندی
grouped = df.groupby("شهر")
print("میانگین حقوق هر شهر:\n", grouped["حقوق"].mean())

print("\nآمار کامل:\n", grouped["حقوق"].agg(["mean", "min", "max", "count"]))
