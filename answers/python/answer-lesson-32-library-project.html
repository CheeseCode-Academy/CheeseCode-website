📝 پاسخ تمرین ۱

سوال: به کلاس کتاب یک ویژگی سال_انتشار اضافه کن.

✅ پاسخ:

class کتاب:
    def __init__(self, عنوان, نویسنده, سال, شابک):
        self.عنوان = عنوان
        self.نویسنده = نویسنده
        self.سال = سال  # سال انتشار
        self.شابک = شابک
        self.امانت_رفته = False
        self.امانت_گیرنده = None
    
    @property
    def سن_کتاب(self):
        from datetime import datetime
        return datetime.now().year - self.سال
    
    def __str__(self):
        وضعیت = "📕 موجود" if not self.امانت_رفته else f"📗 امانت: {self.امانت_گیرنده.نام if self.امانت_گیرنده else 'نامشخص'}"
        return f"{self.عنوان} - {self.نویسنده} ({self.سال}) [{وضعیت}]"

ک = کتاب("صد سال تنهایی", "گابریل گارسیا مارکز", 1967, "978-123")
print(f"سن کتاب: {ک.سن_کتاب} سال")
خروجی (نمونه): 
سن کتاب: 59 سال

📝 پاسخ تمرین ۲

سوال: به کلاس کتابخانه متدی برای گزارش کتاب‌های امانت رفته اضافه کن.

✅ پاسخ:

class کتابخانه:
    # ... کدهای قبلی ...
    
    def کتاب‌های_امانت_رفته(self):
        نتایج = []
        for کتاب in self.کتاب‌ها:
            if کتاب.امانت_رفته:
                نتایج.append(کتاب)
        return نتایج
    
    def گزارش_امانت(self):
        کتاب‌ها = self.کتاب‌های_امانت_رفته()
        if not کتاب‌ها:
            return "هیچ کتابی امانت رفته نیست!"
        
        result = "📗 کتاب‌های امانت رفته:\n"
        for کتاب in کتاب‌ها:
            member_name = کتاب.امانت_گیرنده.نام if کتاب.امانت_گیرنده else "نامشخص"
            result += f"  - {کتاب.عنوان} (امانت: {member_name})\n"
        return result

# استفاده
کتابخانه_من = کتابخانه("کتابخانه مرکزی")

📝 پاسخ تمرین ۳

سوال: یک رابط کاربری با input() بساز که کاربر بتونه با منو، کتاب اضافه کنه، امانت بده و ...

✅ پاسخ:

def منوی_کتابخانه(کتابخانه):
    while True:
        print("\n" + "="*50)
        print(f"📚 {کتابخانه.نام}")
        print("="*50)
        print("1. نمایش کتاب‌ها")
        print("2. اضافه کردن کتاب")
        print("3. امانت کتاب")
        print("4. بازگرداندن کتاب")
        print("5. جستجوی کتاب")
        print("6. ثبت عضو جدید")
        print("7. خروج")
        
        انتخاب = input("\nانتخاب خود را وارد کنید: ")
        
        if انتخاب == "1":
            print(کتابخانه.نمایش_کتاب‌ها())
            
        elif انتخاب == "2":
            عنوان = input("عنوان کتاب: ")
            نویسنده = input("نویسنده: ")
            سال = int(input("سال انتشار: "))
            شابک = input("شابک: ")
            کتاب = کتاب(عنوان, نویسنده, سال, شابک)
            print(کتابخانه.اضافه_کتاب(کتاب))
            
        elif انتخاب == "3":
            شماره = input("شماره عضو: ")
            شابک = input("شابک کتاب: ")
            print(کتابخانه.امانت_کتاب(شماره, شابک))
            
        elif انتخاب == "4":
            شماره = input("شماره عضو: ")
            شابک = input("شابک کتاب: ")
            print(کتابخانه.بازگرداندن_کتاب(شماره, شابک))
            
        elif انتخاب == "5":
            عبارت = input("عبارت جستجو: ")
            نتایج = کتابخانه.جستجو_کتاب(عبارت)
            if نتایج:
                print("نتایج:")
                for کتاب in نتایج:
                    print(f"  - {کتاب}")
            else:
                print("نتیجه‌ای یافت نشد!")
                
        elif انتخاب == "6":
            نام = input("نام عضو: ")
            شماره = input("شماره عضو: ")
            عضو = عضو(نام, شماره)
            print(کتابخانه.ثبت_عضو(عضو))
            
        elif انتخاب == "7":
            print("خروج از برنامه...")
            break
        
        else:
            print("انتخاب نامعتبر!")

# استفاده
کتابخانه_من = کتابخانه("کتابخانه مرکزی")
منوی_کتابخانه(کتابخانه_من)
