📝 پاسخ تمرین ۱

سوال: با BeautifulSoup تمام لینک‌های یک صفحه رو استخراج کن.

✅ پاسخ:

import requests
from bs4 import BeautifulSoup

url = "https://www.python.org"
response = requests.get(url)
soup = BeautifulSoup(response.text, "html.parser")

links = soup.find_all("a")
print(f"تعداد کل لینک‌ها: {len(links)}")

print("\n۱۰ لینک اول:")
for i, link in enumerate(links[:10]):
    href = link.get("href", "بدون لینک")
    text = link.text.strip() or "بدون متن"
    print(f"{i+1}. {text[:30]} -> {href[:50]}")

📝 پاسخ تمرین ۲

سوال: از یک صفحه وب، تیترهای اخبار رو استخراج کن.

✅ پاسخ:

import requests
from bs4 import BeautifulSoup

url = "https://www.python.org"
response = requests.get(url)
soup = BeautifulSoup(response.text, "html.parser")

# استخراج از بخش اخبار
news_items = soup.select(".event-widget li")
print("📰 آخرین اخبار پایتون:")
for item in news_items[:10]:
    text = item.text.strip()
    if text:
        print(f"  • {text}")

# روش با find_all
news = soup.find_all("a", class_="event-link")
print("\n📰 روش دوم:")
for item in news[:5]:
    print(f"  • {item.text.strip()}")

📝 پاسخ تمرین ۳

سوال: یک جدول HTML رو استخراج کن و به DataFrame تبدیل کن.

✅ پاسخ:

import requests
from bs4 import BeautifulSoup
import pandas as pd

# مثال با جدول
html = """


    

        نام
        سن
        شهر
        شغل
    

    

        علی
        ۲۵
        تهران
        برنامه‌نویس
    

    

        زهرا
        ۳۰
        اصفهان
        مهندس
    

    

        رضا
        ۲۲
        شیراز
        دانشجو
    



"""

soup = BeautifulSoup(html, "html.parser")
table = soup.find("table")

# استخراج داده‌ها
data = []
headers = []

# استخراج هدرها
header_row = table.find("tr")
headers = [th.text.strip() for th in header_row.find_all("th")]

# استخراج سطرها
for row in table.find_all("tr")[1:]:
    cols = row.find_all("td")
    data.append([col.text.strip() for col in cols])

# تبدیل به DataFrame
df = pd.DataFrame(data, columns=headers)
print("DataFrame:\n", df)

# استخراج مستقیم از وب
def scrape_table_from_url(url, table_class=None):
    response = requests.get(url)
    soup = BeautifulSoup(response.text, "html.parser")
    
    if table_class:
        table = soup.find("table", class_=table_class)
    else:
        table = soup.find("table")
    
    if not table:
        return None
    
    # استخراج داده...
    return pd.read_html(str(table))[0]  # روش سریع
