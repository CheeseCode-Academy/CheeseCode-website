📝 پاسخ تمرین ۱

سوال: یک دیتابیس با جدول products (نام، قیمت، تعداد) بساز.

✅ پاسخ:

import sqlite3

conn = sqlite3.connect("store.db")
cursor = conn.cursor()

cursor.execute("""
    CREATE TABLE IF NOT EXISTS products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        price REAL,
        quantity INTEGER
    )
""")

conn.commit()
conn.close()
print("جدول products ساخته شد!")

📝 پاسخ تمرین ۲

سوال: به جدول products چند محصول اضافه کن و همه رو بخون.

✅ پاسخ:

import sqlite3

conn = sqlite3.connect("store.db")
cursor = conn.cursor()

# اضافه کردن محصولات
products = [
    ("لپ‌تاپ", 15000000, 10),
    ("موبایل", 8000000, 20),
    ("تبلت", 5000000, 15),
    ("هدفون", 2000000, 30)
]

cursor.executemany("""
    INSERT INTO products (name, price, quantity)
    VALUES (?, ?, ?)
""", products)

conn.commit()

# خواندن محصولات
cursor.execute("SELECT * FROM products")
for product in cursor.fetchall():
    print(f"ID: {product[0]}, نام: {product[1]}, قیمت: {product[2]:,} تومان, تعداد: {product[3]}")

conn.close()

📝 پاسخ تمرین ۳

سوال: یک کلاس کامل برای مدیریت محصولات بساز (اضافه، حذف، به‌روزرسانی، جستجو).

✅ پاسخ:

import sqlite3
from typing import List, Dict, Optional

class ProductManager:
    def __init__(self, db_name: str = "store.db"):
        self.db_name = db_name
        self._init_table()
    
    def _init_table(self):
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("""
                CREATE TABLE IF NOT EXISTS products (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    name TEXT NOT NULL,
                    price REAL,
                    quantity INTEGER
                )
            """)
            conn.commit()
    
    def add_product(self, name: str, price: float, quantity: int) -> int:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("""
                INSERT INTO products (name, price, quantity)
                VALUES (?, ?, ?)
            """, (name, price, quantity))
            conn.commit()
            return cursor.lastrowid
    
    def get_all_products(self) -> List[Dict]:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM products")
            rows = cursor.fetchall()
            return [{"id": r[0], "name": r[1], "price": r[2], "quantity": r[3]} for r in rows]
    
    def get_product(self, product_id: int) -> Optional[Dict]:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM products WHERE id = ?", (product_id,))
            row = cursor.fetchone()
            if row:
                return {"id": row[0], "name": row[1], "price": row[2], "quantity": row[3]}
            return None
    
    def update_product(self, product_id: int, **kwargs) -> bool:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            for key, value in kwargs.items():
                cursor.execute(f"UPDATE products SET {key} = ? WHERE id = ?", (value, product_id))
            conn.commit()
            return True
    
    def delete_product(self, product_id: int) -> bool:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("DELETE FROM products WHERE id = ?", (product_id,))
            conn.commit()
            return True
    
    def search_products(self, search_term: str) -> List[Dict]:
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT * FROM products WHERE name LIKE ?", (f"%{search_term}%",))
            rows = cursor.fetchall()
            return [{"id": r[0], "name": r[1], "price": r[2], "quantity": r[3]} for r in rows]

# استفاده
pm = ProductManager()

# اضافه کردن
pm.add_product("کتاب", 50000, 100)
pm.add_product("خودکار", 5000, 200)

# جستجو
results = pm.search_products("کتاب")
print("نتایج جستجو:", results)

# به‌روزرسانی
pm.update_product(1, price=60000)

# نمایش همه
for product in pm.get_all_products():
    print(product)

# حذف
pm.delete_product(2)
